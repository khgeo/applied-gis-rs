# មេរៀនទី១៤៖ កម្មវិធី Earth Engine និងផែនទីវេប

រហូតមកដល់ពេលនេះ លទ្ធផលរបស់យើងរស់នៅក្នុង Code Editor៖ ស្គ្រីប Console និងផ្ទាំង Tasks ដែលមានតែអ្នកមានគណនី Earth Engine ទេដែលប្រើបាន។ ប៉ុន្តែអ្នកដែលត្រូវការលទ្ធផលពិតប្រាកដ (មន្ត្រីខេត្ត អង្គការ កសិករ និស្សិតផ្សេងទៀត) មិនសរសេរកូដទេ។ ពួកគេត្រូវការ **តំណមួយ** ដែលបើកផែនទី ជ្រើសឆ្នាំ ចុចលើភូមិ ហើយឃើញក្រាប។

មេរៀននេះបង្រៀនការបង្កើត **កម្មវិធី Earth Engine (Earth Engine Apps)** ដោយ ui API៖ widgets ផ្ទាំង និងផែនទី ព្រឹត្តិការណ៍ និង callback ការរក្សាឲ្យកម្មវិធីលឿន ផែនទីពីរភ្ជាប់គ្នា និងការបោះពុម្ព ព្រមទាំងជម្រើសផ្សេងសម្រាប់ចែករំលែកលទ្ធផល។

## ១៤.១. គោលបំណងសិក្សា

បន្ទាប់ពីបញ្ចប់មេរៀន និស្សិតអាច៖

1. ពិពណ៌នារចនាសម្ព័ន្ធកម្មវិធី Earth Engine (ui.root ui.Panel ui.Map) និង widgets សំខាន់ៗ។
2. សរសេរ callback ដែលធ្វើបច្ចុប្បន្នភាពផែនទី និងក្រាប នៅពេលអ្នកប្រើប្ដូរជម្រើស ឬចុចលើផែនទី។
3. ប្រើ evaluate() ជំនួស getInfo() ហើយពន្យល់ហេតុផល។
4. បង្កើតផែនទីពីរភ្ជាប់គ្នា (SplitPanel · Linker) សម្រាប់ការប្រៀបធៀបមុន–ក្រោយ។
5. បោះពុម្ពកម្មវិធី ហើយប្រៀបធៀបជម្រើសចែករំលែកផ្សេង (Export ទៅ QGIS geemap ផែនទីវេប)។

## ១៤.២. ស្ថានភាពបើកមេរៀន៖ ឧបករណ៍តាមដានទន្លេសាប

គណៈកម្មាធិការគ្រប់គ្រងគ្រោះមហន្តរាយខេត្តជុំវិញបឹងទន្លេសាប ចង់បានឧបករណ៍សាមញ្ញមួយ៖ ជ្រើសឆ្នាំ និងខែ ឃើញផ្ទៃទឹកលើផែនទី ឃើញក្រាបផ្ទៃទឹកប្រចាំខែ ហើយចុចលើភូមិណាមួយ ដើម្បីដឹងថាទីនោះលិចទឹកញឹកញាប់ប៉ុណ្ណា។ មន្ត្រីមិនមានគណនី Earth Engine ទេ ហើយប្រើទូរស័ព្ទ ឬកុំព្យូទ័រយួរដៃ។

សំណួរដែលមេរៀននេះឆ្លើយ៖

- តើកម្មវិធី Earth Engine មានរចនាសម្ព័ន្ធយ៉ាងណា?
- តើកម្មវិធីឆ្លើយតបទៅនឹងការចុច និងការជ្រើសរបស់អ្នកប្រើយ៉ាងដូចម្ដេច?
- តើត្រូវធ្វើឲ្យកម្មវិធីលឿន ហើយបោះពុម្ពវាយ៉ាងណា?

### ១៤.៣.១. រចនាសម្ព័ន្ធកម្មវិធី

<figure markdown>
--8<-- "assets/svg/agv/a14-anatomy.svg"
<figcaption>រូបទី១៤.១៖ ផ្ទាំងបញ្ជា + ផែនទី។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - កម្មវិធីភាគច្រើនមានផ្ទាំងបញ្ជា (widgets) នៅម្ខាង និងផែនទីធំនៅម្ខាងទៀត។
    - Widgets ទាំងអស់ជាវត្ថុ ui.* ដែលដាក់ក្នុង ui.Panel។
    - អ្នកប្រើមិនឃើញកូដទេ៖ ពួកគេឃើញតែផ្ទាំង ផែនទី និងក្រាប។


Code Editor មាន **ui.root** ដែលជាផ្ទាំងមេ។ កម្មវិធីភាគច្រើន៖ លុបផ្ទាំងលំនាំដើម បង្កើត **ui.Panel** សម្រាប់បញ្ជា និង **ui.Map** សម្រាប់ផែនទី ហើយដាក់ទាំងពីរក្នុង ui.root៖

```javascript
ui.root.clear();
var map = ui.Map();
map.setOptions('HYBRID');
map.setCenter(104.0, 12.8, 9);
var panel = ui.Panel({style: {width: '340px', padding: '8px'}});
panel.add(ui.Label('ឧបករណ៍តាមដានទន្លេសាប', {fontSize: '20px', fontWeight: 'bold'}));
panel.add(ui.Label('ជ្រើសឆ្នាំ និងខែ ដើម្បីមើលផ្ទៃទឹក។ ចុចលើផែនទី ដើម្បីមើលភាពញឹកញាប់នៃការលិច។'));
ui.root.add(panel);
ui.root.add(map);
```

### ១៤.៣.២. Widgets

<figure markdown>
--8<-- "assets/svg/agv/a14-widgets.svg"
<figcaption>រូបទី១៤.២៖ ប្លុកសាងសង់នៃកម្មវិធី។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - Widget នីមួយៗមាន style (ទំហំ ពណ៌ ពុម្ពអក្សរ) និង callback (មុខងារដែលរត់ពេលអ្នកប្រើធ្វើអ្វីមួយ)។
    - ui.Chart.* ដដែលក្នុង Console អាចដាក់ក្នុង Panel បាន។
    - ពុម្ពអក្សរខ្មែរបង្ហាញបានក្នុងកម្មវិធី ប៉ុន្តែទំហំ និងចន្លោះបន្ទាត់ត្រូវកែតាម style។


Widget នីមួយៗមាន **style** (ទំហំ ពណ៌ ចន្លោះ) និងអាចមាន **callback**៖

```javascript
var yearSel = ui.Select({items: ['2016', '2017', '2018', '2019', '2020', '2021'], value: '2020'});
var monthSl = ui.Slider({min: 1, max: 12, step: 1, value: 10});
var info = ui.Label('ចុចលើផែនទី…');
panel.add(ui.Label('ឆ្នាំ')).add(yearSel).add(ui.Label('ខែ')).add(monthSl).add(info);
```

`panel.add()` ត្រឡប់ panel វិញ ដូច្នេះអាចភ្ជាប់បានជាខ្សែ។ ui.Select ទទួល items ជាអក្សរ ដូច្នេះត្រូវបម្លែងទៅជាលេខ (`parseInt`) មុនប្រើ។

### ១៤.៣.៣. ព្រឹត្តិការណ៍ និង callback

<figure markdown>
--8<-- "assets/svg/agv/a14-events.svg"
<figcaption>រូបទី១៤.៣៖ កម្មវិធីរង់ចាំ ហើយឆ្លើយតប។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - callback គឺជាមុខងារដែលយើងផ្ដល់ឲ្យ widget៖ វារត់តែនៅពេលអ្នកប្រើធ្វើសកម្មភាព។
    - ប្ដូរស្រទាប់ដោយ layers().set() ជំនួស addLayer() ម្ដងទៀត ដើម្បីកុំឲ្យស្រទាប់ជាន់គ្នារាប់សិប។
    - គណនាតែអ្វីដែលត្រូវការ៖ កម្មវិធីដែលរង់ចាំយូរ ធ្វើឲ្យអ្នកប្រើចាកចេញ។


កម្មវិធីមិនរត់ពីលើចុះក្រោមម្ដងហើយបញ្ចប់ទេ៖ វា **រង់ចាំ** ព្រឹត្តិការណ៍ ហើយរត់ callback៖

```javascript
var gswM = ee.ImageCollection('JRC/GSW1_4/MonthlyHistory');           // ១៩៨៤–២០២១ · 0 គ្មានទិន្នន័យ · 1 ដី · 2 ទឹក
function waterOf(y, m) {
  return gswM.filter(ee.Filter.eq('year', y)).filter(ee.Filter.eq('month', m)).first().eq(2).selfMask();
}
function refresh() {
  var y = parseInt(yearSel.getValue(), 10), m = monthSl.getValue();
  map.layers().set(0, ui.Map.Layer(waterOf(y, m), {palette: ['#1565c0']}, 'ទឹក ' + y + '-' + m));
}
yearSel.onChange(refresh);
monthSl.onChange(refresh);
refresh();
```

ប្រើ `map.layers().set(0, …)` ដើម្បីជំនួសស្រទាប់ទី ០ ជំនួសឲ្យ `map.addLayer()` ដែលនឹងបន្ថែមស្រទាប់ថ្មីរាល់ពេលអ្នកប្រើរំកិល។ JRC Monthly History គ្របរហូតដល់ឆ្នាំ ២០២១ ប៉ុណ្ណោះ៖ សម្រាប់ឆ្នាំថ្មីជាងនេះ ប្រើផែនទីទឹក Sentinel-1 ពីមេរៀនទី៧។

### ១៤.៣.៤. getInfo() ឬ evaluate()

<figure markdown>
--8<-- "assets/svg/agv/a14-evaluate.svg"
<figcaption>រូបទី១៤.៤៖ កុំឲ្យកម្មវិធីគាំង។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - getInfo() ទាញលទ្ធផលមកម៉ាស៊ីនភ្ញៀវ ហើយរង់ចាំ៖ ក្នុងកម្មវិធី វាធ្វើឲ្យផ្ទាំងទាំងមូលកក។
    - evaluate() ស្នើលទ្ធផល ហើយបន្តដំណើរការ៖ ពេលចម្លើយមកដល់ callback ធ្វើបច្ចុប្បន្នភាព Label ឬ Chart។
    - បង្ហាញ «កំពុងគណនា…» មុន evaluate ដើម្បីឲ្យអ្នកប្រើដឹងថាកម្មវិធីកំពុងធ្វើការ (មេរៀនទី២៖ client ធៀប server)។


នៅពេលអ្នកប្រើចុចលើផែនទី យើងចង់បង្ហាញលេខមួយ (ឧ. ភាគរយនៃខែដែលលិច)។ លេខនោះគណនានៅម៉ាស៊ីនមេ (មេរៀនទី២)។ **evaluate()** ស្នើលទ្ធផល ហើយបន្តដំណើរការ៖

```javascript
var occ = ee.Image('JRC/GSW1_4/GlobalSurfaceWater').select('occurrence');
map.onClick(function(c) {
  info.setValue('កំពុងគណនា…');
  var pt = ee.Geometry.Point([c.lon, c.lat]);
  map.layers().set(1, ui.Map.Layer(pt, {color: 'red'}, 'ចំណុច'));
  occ.reduceRegion({reducer: ee.Reducer.first(), geometry: pt, scale: 30}).get('occurrence')
    .evaluate(function(v) {
      info.setValue(v === null ? 'មិនដែលមានទឹក (១៩៨៤–២០២១)' : 'មានទឹក ' + v + '% នៃពេលវេលា (១៩៨៤–២០២១)');
    });
});
```

`getInfo()` ធ្វើឲ្យកម្មវិធីទាំងមូលកក រហូតដល់ម៉ាស៊ីនមេឆ្លើយ ហើយបើម៉ាស៊ីនមេយឺត អ្នកប្រើនឹងគិតថាកម្មវិធីខូច។ ក្នុងកម្មវិធី ត្រូវប្រើ evaluate() ជានិច្ច ហើយបង្ហាញ «កំពុងគណនា…» មុន។

    - ក្រាបត្រូវគណនាលើម៉ាស៊ីនមេ៖ ក្រាបដែលមានច្រើនខែ ច្រើនឆ្នាំ អាចយឺត ដូច្នេះគណនាទុកមុនជា Asset បាន។
    - តម្លៃក្នុងក្រាបនេះជាគំរូ ដើម្បីបង្ហាញទម្រង់។


```javascript
var chartBox = ui.Panel();
panel.add(chartBox);
var lake = ee.Geometry.Rectangle([103.3, 12.2, 104.7, 13.4]);
function yearChart(y) {
  var fc = ee.FeatureCollection(ee.List.sequence(1, 12).map(function(m) {
    var img = gswM.filter(ee.Filter.eq('year', y)).filter(ee.Filter.eq('month', m)).first();
    var km2 = ee.Image.pixelArea().divide(1e6).updateMask(img.eq(2))
      .reduceRegion({reducer: ee.Reducer.sum(), geometry: lake, scale: 300, maxPixels: 1e9}).get('area');
    return ee.Feature(null, {month: m, km2: km2});
  }));
  chartBox.widgets().reset([ui.Chart.feature.byFeature(fc, 'month', 'km2')
    .setOptions({title: 'ផ្ទៃទឹក ' + y + ' (គម²)', legend: {position: 'none'}})]);
}
yearSel.onChange(function(v) { refresh(); yearChart(parseInt(v, 10)); });
yearChart(2020);
```

កម្មវិធី Earth Engine មាន ui.Panel សម្រាប់ widgets និង ui.Map សម្រាប់ផែនទី ក្នុង ui.root។ Widgets (Label Select Slider Button Chart) មាន style និង callback ដែលរត់ពេលអ្នកប្រើធ្វើសកម្មភាព។ ប្រើ layers().set() ដើម្បីជំនួសស្រទាប់ evaluate() ដើម្បីកុំឲ្យកម្មវិធីកក និង scale ធំ ឬ Asset ដែលគណនារួច ដើម្បីល្បឿន។

SplitPanel និង Linker ប្រៀបធៀបមុន–ក្រោយ។ កម្មវិធីបោះពុម្ពតាម Cloud Project ហើយបើកបានដោយគ្មានគណនី។ ជម្រើសផ្សេង (Export ទៅ QGIS geemap ផែនទីវេប) សមសម្រាប់អ្នកប្រើ និងគោលបំណងផ្សេងៗ។

!!! abstract "គំនិតស្នូលនៃមេរៀន"
    ការវិភាគដែលមានតែអ្នកវិភាគអាចប្រើ មានផលប៉ះពាល់តិច។ កម្មវិធីសាមញ្ញ លឿន និងស្មោះត្រង់អំពីដែនកំណត់ នាំលទ្ធផលទៅដល់អ្នកសម្រេចចិត្ត។

## ១៤.៩. សំណួររំលឹក និងវាយតម្លៃ

### ក. សំណួរយល់ដឹង

1. ពិពណ៌នារចនាសម្ព័ន្ធ ui.root ui.Panel និង ui.Map នៃកម្មវិធីមួយ។
2. callback ជាអ្វី? ផ្ដល់ឧទាហរណ៍ពីរ។
3. ហេតុអ្វីប្រើ evaluate() ជំនួស getInfo() ក្នុងកម្មវិធី?
4. ហេតុអ្វីប្រើ layers().set() ជំនួស addLayer() ក្នុង callback?
5. ប្រៀបធៀបកម្មវិធី Earth Engine និងការ Export ទៅ QGIS៖ ពេលណាប្រើមួយណា?

### ខ. សំណួរអនុវត្តគោលគំនិត

6. សរសេរ ui.Select សម្រាប់ខេត្ត ៥ ដែលពេលជ្រើស ធ្វើឲ្យផែនទី zoom ទៅខេត្តនោះ។
7. សរសេរ ui.Button «ទាញយក CSV» ដែលបង្ហាញតំណទាញយក ដោយប្រើ `getDownloadURL` និង evaluate។
8. កម្មវិធីរបស់អ្នកចំណាយពេល ៣០ វិនាទីដើម្បីបង្ហាញក្រាប។ ស្នើដំណោះស្រាយបី។
9. សរសេរមុខងារ legend ដែលទទួលបញ្ជីពណ៌ និងឈ្មោះ ហើយត្រឡប់ ui.Panel។
10. ពិពណ៌នាកម្មវិធីមួយ ដោយប្រើមេរៀនណាមួយក្នុងសៀវភៅនេះ ដែលអាចជួយសហគមន៍របស់អ្នក។

### គ. កិច្ចការខ្លី

បន្ថែម ui.DateSlider ទៅកម្មវិធីទន្លេសាប ដែលបង្ហាញរូបភាព Sentinel-1 VV នៃថ្ងៃដែលជ្រើស (± ៦ ថ្ងៃ)។ សរសេរកថាខណ្ឌមួយពីបញ្ហាដែលអ្នកជួប និងរបៀបដោះស្រាយ។
