# មេរៀនទី១១៖ កសិកម្ម និងការធ្វើផែនទីស្រែ

!!! info "ព័ត៌មានមេរៀន"
    **រយៈពេលបង្រៀនដែលស្នើ៖ ៣ ម៉ោង** · មេរៀនទី ១១ ក្នុងចំណោម ១៥

[:material-presentation-play: ស្លាយបង្រៀនមេរៀននេះ](../slides/lesson-11.html){ .md-button target="_blank" }

ស្រូវគឺជាដំណាំសំខាន់បំផុតនៅកម្ពុជា៖ វាគ្របដណ្ដប់ដីកសិកម្មភាគច្រើន ផ្ដល់ស្បៀងដល់ប្រទេស និងជាប្រភពចំណូលរបស់គ្រួសារជាច្រើនលាន។ ដើម្បីតាមដានស្ថានភាពដំណាំ ប៉ាន់ស្មានទិន្នផល និងវាយតម្លៃការខូចខាតពីទឹកជំនន់ ឬគ្រោះរាំងស្ងួត (មេរៀនទី៧–៨) យើងត្រូវដឹងជាមុនថា **ស្រែនៅទីណា**។

បញ្ហាគឺស្រូវវស្សា ត្រូវបានដាំក្នុងពេលដែលពពកគ្របដណ្ដប់ច្រើនបំផុត។ មេរៀននេះបង្រៀនវិធីធ្វើផែនទីស្រែដោយ **Sentinel-1** ដោយប្រើហត្ថលេខាពេលវេលានៃ VH (មេរៀនទី៥) វិធីអុបទិកសម្រាប់រដូវប្រាំង និងការផ្ទៀងផ្ទាត់ផែនទីជាមួយចំណុច និងស្ថិតិ។

---

## ១១.១. គោលបំណងសិក្សា

បន្ទាប់ពីបញ្ចប់មេរៀន និស្សិតអាច៖

1. ពិពណ៌នាប្រតិទិនស្រូវនៅកម្ពុជា ហើយជ្រើសរយៈពេលរូបភាពដែលសមស្រប។
2. ពន្យល់ហត្ថលេខា VH នៃស្រូវ (ទាបពេលស្ទូង ខ្ពស់ពេលស្រូវធំ) ហើយបង្កើតស៊េរី VH ក្នុង Earth Engine។
3. ធ្វើផែនទីស្រែដោយក្បួនកម្រិត (អប្បបរមា + ការកើនឡើង) និងដោយ Random Forest លើ composite ប្រចាំខែ។
4. ប្រើវិធីអុបទិក (LSWI និង EVI) សម្រាប់ស្រូវប្រាំង ហើយប្រៀបធៀបជាមួយ SAR។
5. ផ្ទៀងផ្ទាត់ផែនទីស្រែដោយចំណុច និងស្ថិតិ ហើយរាយការណ៍ដែនកំណត់។

## ១១.២. ស្ថានភាពបើកមេរៀន៖ ផ្ទៃស្រូវវស្សាខេត្តតាកែវ

មន្ទីរកសិកម្មខេត្តតាកែវ ចង់ផ្ទៀងផ្ទាត់ផ្ទៃស្រូវវស្សាឆ្នាំ ២០២៤ ដែលរាយការណ៍ដោយស្រុក ហើយចង់ដឹងថាស្រែណាខ្លះដែលមិនត្រូវបានដាំ (ទុកទំនេរ) ក្នុងឆ្នាំនេះ។ រូបភាព Sentinel-2 ក្នុងខែកក្កដា–តុលា ស្ទើរតែទាំងអស់មានពពក។ ក្រុមរបស់អ្នកមានចំណុចស្រែ និងមិនមែនស្រែ ដែលមន្ត្រីកសិកម្មបានប្រមូលក្នុងការចុះវាល។

សំណួរដែលមេរៀននេះឆ្លើយ៖

- តើស្រូវមានលក្ខណៈអ្វីក្នុងរូបភាពរ៉ាដា ដែលដំណាំ និងគម្របដីផ្សេងទៀតគ្មាន?
- តើក្បួនសាមញ្ញ ឬ classifier ល្អជាង?
- តើផែនទីស្រែត្រូវផ្ទៀងផ្ទាត់ដោយរបៀបណា?

## ១១.៣. ទ្រឹស្ដីស្នូល

### ១១.៣.១. ប្រតិទិនស្រូវ

<figure markdown>
--8<-- "assets/svg/agv/a11-calendar.svg"
<figcaption>រូបទី១១.១៖ ស្រូវច្រើនប្រភេទ ច្រើនរដូវ។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - ស្រូវវស្សា គ្របដណ្ដប់ផ្ទៃធំជាងគេ ហើយដាំក្នុងពេលពពកច្រើនបំផុត។
    - ស្រូវប្រាំង និងស្រូវទឹកស្រក នៅតំបន់ធារាសាស្ត្រ និងជុំវិញបឹង៖ អុបទិកមើលឃើញបានល្អជាង។
    - ប្រតិទិននេះប្រហាក់ប្រហែល៖ ត្រូវកែតាមខេត្ត ដើម្បីជ្រើសរយៈពេលស្ទូងដែលត្រឹមត្រូវ។


កម្ពុជាមានប្រព័ន្ធស្រូវច្រើនប្រភេទ៖ **ស្រូវវស្សា** (ស្ទូងមិថុនា–សីហា ច្រូតវិច្ឆិកា–ធ្នូ) ដែលគ្របផ្ទៃធំជាងគេ **ស្រូវវស្សាដំបូង** (ស្រូវខ្លី) **ស្រូវប្រាំង** ក្នុងតំបន់ធារាសាស្ត្រ និង **ស្រូវទឹកស្រក** ជុំវិញបឹងទន្លេសាប និងតាមដងទន្លេ។ ពេលវេលាខុសគ្នាតាមខេត្ត ពូជ និងភ្លៀងក្នុងឆ្នាំនីមួយៗ។ ដូច្នេះ មុនធ្វើផែនទី ត្រូវកំណត់ **រយៈពេលស្ទូង** និង **រយៈពេលលូតលាស់** សម្រាប់តំបន់សិក្សា ដោយប្រើប្រតិទិនក្នុងស្រុក ឬស៊េរី NDVI/VH នៃស្រែដែលដឹង។

### ១១.៣.២. ហត្ថលេខា VH របស់ស្រូវ

<figure markdown>
--8<-- "assets/svg/agv/a11-vh.svg"
<figcaption>រូបទី១១.២៖ ទាបពេលស្ទូង ខ្ពស់ពេលស្រូវធំ។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - ពេលស្ទូង វាលស្រែមានទឹក៖ VH ធ្លាក់ទាបដូចទឹក (−២២ ដល់ −២៥ dB)។
    - ពេលស្រូវលូតលាស់ ដើមស្រូវបង្កើនការចាំងត្រឡប់៖ VH ឡើង ៥–៩ dB។
    - ទឹកអចិន្ត្រៃយ៍ទាបជានិច្ច ព្រៃ និងទីក្រុងខ្ពស់ជានិច្ច៖ មិនមានទម្រង់ «ចុះ រួចឡើង» ទេ។


ស្រូវមានហត្ថលេខារ៉ាដាដែលពិសេស៖

1. **មុនស្ទូង**៖ ដីភ្ជួរ ឬស្រែស្ងួត VH ប្រហែល −១៦ ដល់ −១៨ dB។
2. **ពេលបញ្ចូលទឹក និងស្ទូង**៖ វាលស្រែជាផ្ទៃទឹករលោង VH ធ្លាក់ដល់ −២២ ដល់ −២៥ dB (ដូចមេរៀនទី៧)។
3. **ពេលស្រូវលូតលាស់**៖ ដើម និងស្លឹកស្រូវបង្កើតការចាំងត្រឡប់ច្រើនដង VH ឡើង ៥–៩ dB ក្នុងរយៈពេល ១–២ ខែ។
4. **ពេលទុំ និងច្រូត**៖ VH ថយបន្តិច ហើយស្ថិតលំនឹង។

លំនាំ **«ចុះ រួចឡើង»** នេះហើយដែលបែងចែកស្រូវ ពីទឹកអចិន្ត្រៃយ៍ (ទាបជានិច្ច) ពីព្រៃ និងទីក្រុង (ខ្ពស់ជានិច្ច) និងពីដំណាំដីទួល (មិនដែលធ្លាក់ទាបដូចទឹក)។ VH ប្រតិកម្មទៅនឹងជីវម៉ាសរុក្ខជាតិខ្លាំងជាង VV ដូច្នេះវាជាប៉ូលការីសាស្យុងចម្បងសម្រាប់ស្រូវ។

```javascript
var aoi = ee.FeatureCollection('FAO/GAUL/2015/level1').filter(ee.Filter.eq('ADM1_NAME', 'Takeo'));
var s1 = ee.ImageCollection('COPERNICUS/S1_GRD').filterBounds(aoi)
  .filter(ee.Filter.eq('instrumentMode', 'IW'))
  .filter(ee.Filter.listContains('transmitterReceiverPolarisation', 'VH'))
  .filter(ee.Filter.eq('orbitProperties_pass', 'DESCENDING'))
  .filterDate('2024-04-01', '2025-01-15').select('VH');
var paddy = ee.Geometry.Point([104.79, 10.99]);         // ស្រែដែលដឹង (ប្ដូរ)
print(ui.Chart.image.series(s1, paddy.buffer(50), ee.Reducer.mean(), 10).setOptions({title: 'VH ស្រែ'}));
```

ប្រើ buffer ៥០ ម ជុំវិញចំណុច ដើម្បីកាត់បន្ថយ speckle ក្នុងក្រាប។

### ១១.៣.៣. ក្បួនកម្រិត៖ អប្បបរមា និងការកើនឡើង

<figure markdown>
--8<-- "assets/svg/agv/a11-rules.svg"
<figcaption>រូបទី១១.៣៖ លក្ខខណ្ឌពីរ បែងចែកស្រែបាន។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - ទឹកអចិន្ត្រៃយ៍មាន min ទាប ប៉ុន្តែមិនឡើងវិញ៖ លក្ខខណ្ឌទីពីរកាត់វាចេញ។
    - ដំណាំដទៃ និងព្រៃមិនធ្លាក់ដល់ −១៩ dB៖ លក្ខខណ្ឌទីមួយកាត់វាចេញ។
    - កម្រិតក្នុងរូបនេះជាឧទាហរណ៍៖ ត្រូវកែតាមតំបន់ ដោយប្រើចំណុចស្រែដែលដឹង។


ដោយប្រើ composite ១២ ថ្ងៃ (ឬ median ក្នុងរយៈពេល ១២ ថ្ងៃ) យើងគណនាលក្ខណៈពីរក្នុងក្រឡានីមួយៗ៖

```javascript
var start = ee.Date('2024-05-01');
var steps = ee.List.sequence(0, 17);                          // ១៨ × ១២ ថ្ងៃ ≈ ៧ ខែ
var series = ee.ImageCollection.fromImages(steps.map(function(k) {
  var d = start.advance(ee.Number(k).multiply(12), 'day');
  var sub = s1.filterDate(d, d.advance(12, 'day'));
  var empty = ee.Image.constant(0).rename('VH').updateMask(0);      // ចន្លោះដែលគ្មានការថត
  return ee.Image(ee.Algorithms.If(sub.size().gt(0), sub.median().focalMedian(30, 'circle', 'meters'), empty))
    .set('system:time_start', d.millis());
}));
var transplant = series.filterDate('2024-06-01', '2024-09-15');   // រយៈពេលស្ទូង
var growth = series.filterDate('2024-07-15', '2024-11-15');       // រយៈពេលលូតលាស់
var vhMin = transplant.min();
var rise = growth.max().subtract(vhMin);
var rice = vhMin.lt(-19).and(rise.gt(5)).rename('rice');
Map.addLayer(rice.selfMask().clip(aoi), {palette: ['#2e7d32']}, 'ស្រែ (ក្បួន)');
```

`ee.Algorithms.If` ជំនួស composite ដែលគ្មានរូបភាព (ចន្លោះ ១២ ថ្ងៃដែលគ្មានការថត) ដោយរូបភាព VH ទទេ ដូចក្នុងមេរៀនទី៥ ដើម្បីកុំឲ្យ min() និង max() បរាជ័យ។ កម្រិត −១៩ dB និង ៥ dB ជាចំណុចចាប់ផ្ដើម៖ ត្រូវកែតាមតំបន់ ដោយមើលហ៊ីស្តូក្រាមនៃ `vhMin` និង `rise` នៅចំណុចស្រែ និងមិនមែនស្រែដែលដឹង។

    ប្ដូរកម្រិតអប្បបរមា និងការកើនឡើង ហើយមើលថាក្រឡាប្រភេទណាត្រូវបានចាត់ជាស្រែ។ ក្រឡាក្នុងពិសោធន៍ត្រូវបានក្លែងធ្វើតាមលំនាំ VH នៃស្រែ ដំណាំដទៃ ព្រៃ ទីក្រុង និងទឹក។

### ១១.៣.៤. Composite ប្រចាំខែ និង Random Forest

<figure markdown>
--8<-- "assets/svg/agv/a11-rgb.svg"
<figcaption>រូបទី១១.៤៖ មើលស្រែក្នុងរូបភាពតែមួយ។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - ដាក់ VH បីខែជា RGB៖ ពណ៌ប្រាប់ពីការប្ដូរតាមពេលវេលា មិនមែនពណ៌ពិតទេ។
    - ស្រែដែលស្ទូងយឺត ឬលឿន បង្ហាញពណ៌ខុសគ្នា (ក្រហម បៃតង ខៀវ)។
    - Composite ប្រចាំខែ ក៏អាចជាលក្ខណៈសម្រាប់ Random Forest (មេរៀនទី៦)។


ក្បួនកម្រិតងាយយល់ ប៉ុន្តែពិបាកកែនៅពេលស្រែមានប្រតិទិនខុសគ្នាច្រើន។ ជម្រើសមួយទៀតគឺប្រើ VH ប្រចាំខែជា **លក្ខណៈ** ហើយបណ្ដុះបណ្ដាល Random Forest (មេរៀនទី៦)៖

```javascript
var months = ee.List.sequence(5, 11);
var stack = ee.ImageCollection.fromImages(months.map(function(m) {
  var d = ee.Date.fromYMD(2024, m, 1);
  return s1.filterDate(d, d.advance(1, 'month')).median();
})).toBands().rename(months.map(function(m) { return ee.String('VH_').cat(ee.Number(m).format('%02d')); }));
Map.addLayer(stack.select(['VH_07', 'VH_08', 'VH_10']).clip(aoi), {min: -25, max: -10}, 'VH កក្កដា សីហា តុលា');
// training = ចំណុចស្រែ (1) និងមិនមែនស្រែ (0) ដែលមាន property 'rice'
```

`toBands()` ដាក់ឈ្មោះក្រុមរលកដោយលេខរៀង ដូច្នេះយើង rename វាវិញ។ Random Forest រៀនទម្រង់ពេលវេលាដោយខ្លួនឯង រួមទាំងស្រែដែលស្ទូងយឺត ឬលឿន។ ប៉ុន្តែវាត្រូវការចំណុចគំរូច្រើន ដែលតំណាងប្រតិទិនទាំងអស់ក្នុងតំបន់។

### ១១.៣.៥. វិធីអុបទិក៖ ស្រូវប្រាំង

<figure markdown>
--8<-- "assets/svg/agv/a11-optical.svg"
<figcaption>រូបទី១១.៥៖ ទឹកមុន បៃតងក្រោយ (អុបទិក)។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - ក្នុងរយៈពេលស្ទូង LSWI ឡើងជិត ឬលើស EVI៖ ផ្ទៃមានទឹក និងរុក្ខជាតិតិច។
    - បន្ទាប់ពីសញ្ញាលិចទឹក EVI ត្រូវឡើងខ្ពស់ក្នុងរយៈពេល ២–៣ ខែ ដើម្បីបញ្ជាក់ថាជាស្រូវ។
    - វិធីនេះល្អសម្រាប់ស្រូវប្រាំង ប៉ុន្តែក្នុងរដូវវស្សា ពពកធ្វើឲ្យខកសញ្ញាលិចទឹកញឹកញាប់។


សម្រាប់ស្រូវប្រាំង (ដែលមានពពកតិច) វិធី **Xiao et al. (២០០៥)** ប្រើ MODIS ឬ Sentinel-2 ដើម្បីរកសញ្ញាលិចទឹក៖ ក្រឡាដែល **LSWI + ០,០៥ ≥ EVI** (ឬ NDVI) ក្នុងរយៈពេលស្ទូង ហើយបន្ទាប់មក EVI ឡើងខ្ពស់ក្នុងរយៈពេល ២–៣ ខែ៖

```javascript
function idx(img) {
  var evi = img.expression('2.5 * (N - R) / (N + 6 * R - 7.5 * B + 1)',
    {N: img.select('B8'), R: img.select('B4'), B: img.select('B2')}).rename('EVI');
  var lswi = img.normalizedDifference(['B8', 'B11']).rename('LSWI');
  return evi.addBands(lswi).copyProperties(img, ['system:time_start']);
}
// s2 = Sentinel-2 ដែលបានលុបពពក និង divide(10000) (មេរៀនទី៣)
var flood = s2.filterDate('2024-12-01', '2025-02-01').map(idx).map(function(i) {
  return i.select('LSWI').add(0.05).gte(i.select('EVI'));
}).max();
var green = s2.filterDate('2025-02-01', '2025-04-15').map(idx).select('EVI').max().gt(0.45);
var dryRice = flood.and(green);
```

ក្នុងរដូវវស្សា ពពកធ្វើឲ្យខកសញ្ញាលិចទឹកញឹកញាប់ ដូច្នេះវិធីអុបទិក និង SAR គួរតែប្រើរួមគ្នា៖ SAR សម្រាប់វស្សា អុបទិកសម្រាប់ប្រាំង ឬលក្ខណៈទាំងពីរក្នុង Random Forest តែមួយ។

### ១១.៣.៦. Mask និងការសម្អាត

ដកចេញ៖ ទឹកអចិន្ត្រៃយ៍ (JRC · មេរៀនទី៧) តំបន់ជម្រាលលើស ៥° (ស្រមោលរ៉ាដា និងមិនមែនស្រែ) និងទីក្រុង (WorldCover ៥០)។ បន្ទាប់មកដកក្រុមក្រឡាតូចៗ ដោយ `connectedPixelCount()` ដើម្បីកាត់ speckle ដែលនៅសល់។

!!! note "អានរូបនេះ"
    - ភាពត្រឹមត្រូវតាមចំណុច វាស់ថាផែនទីដាក់ស្លាកត្រឹមត្រូវប៉ុណ្ណា នៅកន្លែងដែលមានចំណុច។
    - ការប្រៀបធៀបផ្ទៃតាមខេត្តជាមួយស្ថិតិក្រសួងកសិកម្ម បង្ហាញលំអៀងជាប្រព័ន្ធ (ច្រើន ឬតិចពេក)។
    - ស្ថិតិក៏មានកំហុសដែរ៖ ភាពខុសគ្នាធំ ត្រូវពិនិត្យទាំងពីរផ្នែក។ ចំណុចក្នុងក្រាបជាតម្លៃគំរូ។


1. **តាមចំណុច**៖ តារាងកំហុស ២ × ២ (ស្រែ / មិនមែនស្រែ) ជាមួយ PA UA និង OA (មេរៀនទី៦) ពីចំណុចដែលមិនបានប្រើក្នុងការកែកម្រិត ឬបណ្ដុះបណ្ដាល។
2. **តាមផ្ទៃ**៖ ផ្ទៃស្រូវតាមស្រុក ឬខេត្ត ធៀបនឹងស្ថិតិក្រសួងកសិកម្ម។ គណនាផ្ទៃដោយ pixelArea ហើយគូរក្រាប scatter (ផែនទី ធៀបនឹងស្ថិតិ)។

ភាពខុសគ្នាជាប្រព័ន្ធ (ឧ. ផែនទីតិចជាងស្ថិតិ ២០% គ្រប់ខេត្ត) បង្ហាញលំអៀង៖ ប្រហែលជាស្រែតូចៗ និងភ្លឺស្រែ ត្រូវបានខក។ ភាពខុសគ្នាធំតែនៅខេត្តមួយ ត្រូវពិនិត្យទាំងផែនទី និងស្ថិតិ។

## ១១.៤. ឧទាហរណ៍ដែលបានដោះស្រាយ៖ ស្រូវវស្សា ២០២៤ ខេត្តតាកែវ

### ជំហានទី១៖ ពិនិត្យស្រែដែលដឹង

ដាក់ចំណុចស្រែបីកន្លែងដែលអ្នកដឹង ហើយទាញក្រាប VH (ផ្នែក ១១.៣.២)។ កំណត់ពេលធ្លាក់ទាបបំផុត និងពេលឡើងខ្ពស់ ដើម្បីកែ `transplant` និង `growth`។

### ជំហានទី២៖ ស៊េរី និងលក្ខណៈ

បង្កើត `series` `vhMin` និង `rise` (ផ្នែក ១១.៣.៣)។ បង្ហាញ `vhMin` និង `rise` ជាផែនទី ហើយពិនិត្យតម្លៃនៅស្រែ ភូមិ ព្រៃ និងបឹងដោយ Inspector។

### ជំហានទី៣៖ កម្រិត mask និងផ្ទៃ

```javascript
var jrc = ee.Image('JRC/GSW1_4/GlobalSurfaceWater').select('seasonality').unmask(0);
var slope = ee.Terrain.slope(ee.Image('USGS/SRTMGL1_003'));
var wc = ee.ImageCollection('ESA/WorldCover/v200').first();
var riceClean = rice.updateMask(jrc.lt(10)).updateMask(slope.lt(5)).updateMask(wc.neq(50));
riceClean = riceClean.updateMask(riceClean.connectedPixelCount(20, true).gte(10)).selfMask();
var ha = ee.Image.pixelArea().divide(1e4).updateMask(riceClean).reduceRegions({
  collection: ee.FeatureCollection('FAO/GAUL/2015/level2').filter(ee.Filter.eq('ADM1_NAME', 'Takeo')),
  reducer: ee.Reducer.sum(), scale: 10, tileScale: 8});
print(ha.select(['ADM2_NAME', 'sum']));
```

### ជំហានទី៤៖ ផ្ទៀងផ្ទាត់

ប្រើចំណុចស្រែ និងមិនមែនស្រែ (property `rice` = ១ ឬ ០)៖

```javascript
var cm = riceClean.unmask(0).rename('classification').sampleRegions({collection: points, properties: ['rice'], scale: 10})

print('OA', cm.accuracy(), 'PA', cm.producersAccuracy(), 'UA', cm.consumersAccuracy());
```

    ក្រឡា ១០ ម មិនអាចបែងចែកភ្លឺស្រែ និងផ្លូវលំតូចៗទេ ដូច្នេះផ្ទៃស្រែពីផែនទីអាចខុសពីផ្ទៃដែលដាំពិត។ ស្រែដែលមិនបានបញ្ចូលទឹកមុនស្ទូង (ស្រូវព្រោះ ឬស្រែទួល) អាចមិនបង្ហាញការធ្លាក់ VH ច្បាស់ ហើយត្រូវបានខក។

## ១១.៥. សកម្មភាពអនុវត្ត

- **លំហាត់ទី១១**៖ ធ្វើផែនទីស្រូវវស្សាពី Sentinel-1 សម្រាប់ខេត្តរបស់អ្នក ដោយក្បួនកម្រិត និង Random Forest ប្រៀបធៀបភាពត្រឹមត្រូវ និងផ្ទៃ ហើយនាំចេញ។ មើល [លំហាត់ទី១១](../workbook/lab-11.md)។
- **ក្នុងថ្នាក់**៖ និស្សិតទាញក្រាប VH នៃស្រែនៅភូមិកំណើត ហើយប្រៀបធៀបពេលស្ទូងជាមួយអ្វីដែលគ្រួសារបានធ្វើ។

---

## ១១.៦. ការយល់ច្រឡំដែលត្រូវប្រុងប្រយ័ត្ន

**១. «ស្រូវត្រូវបានរកឃើញដោយ NDVI ខ្ពស់»**
ដំណាំ និងរុក្ខជាតិជាច្រើនមាន NDVI ខ្ពស់។ ស្រូវខុសគេដោយ «ទឹកមុន បៃតងក្រោយ» មិនមែនដោយភាពបៃតងទេ។

**២. «កម្រិតមួយប្រើបានគ្រប់ខេត្ត»**
ប្រតិទិន ពូជ និងប្រភេទដីខុសគ្នា។ កម្រិត និងរយៈពេលត្រូវកែតាមតំបន់ ដោយប្រើស្រែដែលដឹង។

**៣. «VV និង VH ល្អដូចគ្នាសម្រាប់ស្រូវ»**
VH ប្រតិកម្មទៅនឹងជីវម៉ាសរុក្ខជាតិខ្លាំងជាង ហើយបង្ហាញការកើនឡើងច្បាស់ជាង។ VV ល្អសម្រាប់ទឹក។

**៤. «ផ្ទៃស្រែពីផែនទីគួរតែស្មើស្ថិតិ»**
ស្ថិតិ និងផែនទីមាននិយមន័យ និងកំហុសខុសគ្នា។ ភាពខុសគ្នា ១០–២០% ជារឿងធម្មតា ហើយត្រូវពន្យល់។

**៥. «ផែនទីស្រែមួយឆ្នាំ ប្រើបានរៀងរហូត»**
ស្រែខ្លះត្រូវទុកទំនេរ ឬប្ដូរទៅដំណាំផ្សេង ឬក្លាយជាដីសង់ផ្ទះ។ ផែនទីត្រូវធ្វើរៀងរាល់ឆ្នាំ។

---

## ១១.៧. សេចក្ដីសង្ខេបមេរៀន

ស្រូវវស្សាដាំក្នុងរដូវពពក ដូច្នេះ Sentinel-1 ជាប្រភពចម្បង។ VH នៃស្រែធ្លាក់ទាបពេលបញ្ចូលទឹក និងស្ទូង ហើយឡើង ៥–៩ dB ពេលស្រូវលូតលាស់៖ លំនាំ «ចុះ រួចឡើង» នេះបែងចែកស្រូវពីទឹក ព្រៃ ទីក្រុង និងដំណាំដីទួល។ ក្បួនកម្រិត (VH អប្បបរមា និងការកើនឡើង) ងាយយល់ ខណៈ Random Forest លើ VH ប្រចាំខែ សម្របនឹងប្រតិទិនខុសគ្នាបានល្អជាង។

វិធីអុបទិក (LSWI + ០,០៥ ≥ EVI) ល្អសម្រាប់ស្រូវប្រាំង។ Mask ទឹក ជម្រាល និងទីក្រុង ហើយការផ្ទៀងផ្ទាត់ពីរកម្រិត (ចំណុច និងស្ថិតិ) ធ្វើឲ្យផែនទីទុកចិត្តបាន។

!!! abstract "គំនិតស្នូលនៃមេរៀន"
    ស្រូវត្រូវបានរកឃើញដោយអ្វីដែលវាធ្វើតាមពេលវេលា មិនមែនដោយរូបរាងក្នុងថ្ងៃតែមួយទេ៖ ទឹកមុន បៃតងក្រោយ។ រ៉ាដាមើលឃើញលំនាំនេះ សូម្បីតែក្រោមពពក។

## ១១.៩. សំណួររំលឹក និងវាយតម្លៃ

### ក. សំណួរយល់ដឹង

1. ពិពណ៌នាដំណាក់កាលបួននៃហត្ថលេខា VH នៃស្រូវ។
2. ហេតុអ្វី VH ល្អជាង VV សម្រាប់ធ្វើផែនទីស្រូវ?
3. ហេតុអ្វីក្បួនត្រូវការលក្ខខណ្ឌទាំងពីរ (អប្បបរមា និងការកើនឡើង)?
4. ពន្យល់វិធី LSWI + ០,០៥ ≥ EVI ហើយប្រាប់ពេលណាវាមិនដំណើរការ។
5. ហេតុអ្វីភាពត្រឹមត្រូវតាមចំណុចល្អ មិនធានាថាផ្ទៃត្រឹមត្រូវ?

### ខ. សំណួរអនុវត្តគោលគំនិត

6. សរសេរកូដគណនា VH ប្រចាំខែ ពីឧសភា ដល់វិច្ឆិកា ហើយបង្ហាញជាក្រាបសម្រាប់ចំណុចបី។
7. ផែនទីរបស់អ្នកចាត់បឹង និងស្រះជាស្រែ។ មូលហេតុ និងដំណោះស្រាយ?
8. ផែនទីមានផ្ទៃស្រូវតិចជាងស្ថិតិ ៣០% នៅស្រុកមួយ។ រាយមូលហេតុបីដែលអាចកើតឡើង។
9. ស្នើរបៀបរួមបញ្ចូល VH និង NDVI ក្នុង Random Forest តែមួយ។
10. តើផែនទីស្រែ ជួយការវាយតម្លៃគ្រោះរាំងស្ងួត (មេរៀនទី៨) យ៉ាងដូចម្ដេច?

### គ. កិច្ចការខ្លី

ទាញក្រាប VH នៃស្រែចំនួនប្រាំ ពីស្រុកខុសគ្នាក្នុងខេត្តរបស់អ្នក។ ធ្វើតារាងពេលស្ទូង (VH ទាបបំផុត) និងពេលកំពូល ហើយសរសេរកថាខណ្ឌមួយពីភាពខុសគ្នា និងផលប៉ះពាល់លើការជ្រើសរយៈពេលក្នុងក្បួន។

## កំណត់សម្គាល់សម្រាប់គ្រូបង្រៀន

- ស៊េរី VH និងក្រឡាក្នុងរូប និងពិសោធន៍ ត្រូវបានក្លែងធ្វើតាមលំនាំដែលបានរាយការណ៍សម្រាប់ស្រែអាស៊ីអាគ្នេយ៍។ និស្សិតទាញស៊េរីពិតក្នុងលំហាត់ទី១១។
- ប្រសិនបើគ្មានចំណុចវាល និស្សិតអាចបង្កើតចំណុចផ្ទៀងផ្ទាត់ពីរូបភាព Google Earth និងក្រាប VH ប៉ុន្តែត្រូវរាយការណ៍ថាវាមិនមែនការស្ទង់វាល។

### គ. ការភ្ជាប់ទៅមេរៀនបន្ទាប់

ស្រែជាច្រើនជុំវិញភ្នំពេញ និងទីរួមខេត្ត កំពុងក្លាយជាបុរី និងរោងចក្រ។ ការប្ដូរនេះមិនត្រឹមតែកាត់បន្ថយដីកសិកម្មទេ ប៉ុន្តែក៏ប្ដូរសីតុណ្ហភាពផ្ទៃ និងហានិភ័យទឹកជំនន់ផងដែរ។
