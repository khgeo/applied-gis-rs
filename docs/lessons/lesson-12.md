# មេរៀនទី១២៖ ទីក្រុង ការពង្រីក និងសីតុណ្ហភាពផ្ទៃដី

!!! info "ព័ត៌មានមេរៀន"
    **រយៈពេលបង្រៀនដែលស្នើ៖ ៣ ម៉ោង** · មេរៀនទី ១២ ក្នុងចំណោម ១៥

[:material-presentation-play: ស្លាយបង្រៀនមេរៀននេះ](../slides/lesson-12.html){ .md-button target="_blank" }

ភ្នំពេញ និងទីរួមខេត្តធំៗ កំពុងពង្រីកយ៉ាងលឿន៖ បុរីថ្មី រោងចក្រ ផ្លូវល្បឿនលឿន និងការចាក់ដីបំពេញបឹង និងវាលស្រែជុំវិញ។ ក្រុងព្រះសីហនុបានប្ដូររូបរាងស្ទើរទាំងស្រុងក្នុងរយៈពេលតែប៉ុន្មានឆ្នាំ។ ការពង្រីកនេះប៉ះពាល់ដល់ដីកសិកម្ម ផ្លូវទឹក និងហានិភ័យទឹកជំនន់ ហើយវាធ្វើឲ្យទីក្រុង **ក្ដៅជាងជនបទជុំវិញ**។

មេរៀននេះបង្រៀនការវាស់ការពង្រីកទីក្រុង ដោយសន្ទស្សន៍ និងទិន្នន័យសកល (GHSL Dynamic World WorldCover ពន្លឺពេលយប់) និងការគណនា **សីតុណ្ហភាពផ្ទៃដី (LST)** ពី Landsat ដើម្បីវិភាគកោះកំដៅទីក្រុង និងការប៉ះពាល់កំដៅ។

---

## ១២.១. គោលបំណងសិក្សា

បន្ទាប់ពីបញ្ចប់មេរៀន និស្សិតអាច៖

1. រកសំណង់ និងដីទទេថ្មី ដោយការប្ដូរ NDVI និង NDBI រវាងពីរឆ្នាំ ហើយពន្យល់ដែនកំណត់។
2. ប្រើ GHSL Dynamic World WorldCover និង VIIRS ដើម្បីវាស់ការពង្រីកទីក្រុង និងជ្រើសទិន្នន័យតាមសំណួរ។
3. គណនា LST ជា °C ពី Landsat Collection 2 Level-2 (ST_B10) ជាមួយការលុបពពក។
4. វាស់កោះកំដៅទីក្រុង (UHI) និងទំនាក់ទំនង LST–NDVI។
5. គណនាស្ថិតិកំដៅតាមខណ្ឌ ហើយភ្ជាប់ជាមួយប្រជាជន ដើម្បីកំណត់តំបន់អាទិភាព។

## ១២.២. ស្ថានភាពបើកមេរៀន៖ ភ្នំពេញក្នុងរដូវក្ដៅ

សាលារាជធានីភ្នំពេញ ចង់ដាំដើមឈើ និងបង្កើតឧទ្យានតូចៗ នៅតំបន់ដែលក្ដៅបំផុត និងមានប្រជាជនរស់នៅក្រាស់។ ពួកគេក៏ចង់ដឹងថាទីក្រុងបានពង្រីកប៉ុណ្ណាចាប់ពីឆ្នាំ ២០១៦ ហើយតំបន់ពង្រីកថ្មី ក្ដៅជាងតំបន់ដែលនៅតែជាស្រែឬទេ។ ស្ថានីយឧតុនិយមមានតែពីរបីកន្លែង មិនអាចបង្ហាញភាពខុសគ្នាក្នុងទីក្រុងបានទេ។

សំណួរដែលមេរៀននេះឆ្លើយ៖

- តើត្រូវវាស់ផ្ទៃទីក្រុង និងការពង្រីករបស់វាយ៉ាងដូចម្ដេច?
- តើ LST ពីផ្កាយរណបគណនាយ៉ាងណា ហើយវាខុសពីសីតុណ្ហភាពខ្យល់យ៉ាងណា?
- តើខណ្ឌណាក្ដៅបំផុត ហើយមានប្រជាជនប៉ុន្មាននៅទីនោះ?

!!! note "អានរូបនេះ"
    - ក្រុងព្រះសីហនុបានសាងសង់ច្រើនក្នុងរយៈពេល ២០១៦–២០២០។ រូបភាពពីរឆ្នាំបង្ហាញដីទទេ និងអគារថ្មី។
    - ទីក្រុងជាគម្របដីដែលប្ដូរលឿនបំផុត ហើយការប្ដូរនេះកម្របញ្ច្រាសវិញ។
    - រូបភាពពិត Sentinel-2 ដែលបានពង្រីកពន្លឺតាមក្រុមរលក សម្រាប់ការបង្ហាញ។


ទីក្រុងជាគម្របដីដែលប្ដូរលឿន និងកម្របញ្ច្រាសវិញ។ ក្នុងរូបភាពអុបទិក ការពង្រីកបង្ហាញជា៖ រុក្ខជាតិបាត់ (NDVI ធ្លាក់) ដីទទេ និងដីចាក់បំពេញ (ភ្លឺក្នុង SWIR) បន្ទាប់មកអគារ និងផ្លូវ (NDBI ខ្ពស់)។ ដំណាក់កាល «ដីទទេ» អាចមានរយៈពេលច្រើនឆ្នាំ ជាពិសេសនៅបុរីដែលចាក់ដីរួច ប៉ុន្តែមិនទាន់សាងសង់។

### ១២.៣.២. រកសំណង់ថ្មីដោយសន្ទស្សន៍

<figure markdown>
--8<-- "assets/svg/agv/a12-change.svg"
<figcaption>រូបទី១២.២៖ លក្ខខណ្ឌបី ដើម្បីកាត់កំហុស។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - NDVI ធ្លាក់តែមួយ អាចជាដំណាំច្រូត ឬរដូវខុសគ្នា៖ ត្រូវការ NDVI ទាបក្នុងឆ្នាំថ្មី និង NDBI ឡើង។
    - NDBI នៅកម្ពុជាច្រឡំរវាងដីទទេ និងអគារ៖ ផ្ទៃនេះរួមទាំងដីចាក់បំពេញដែលមិនទាន់សាងសង់។
    - តួលេខគណនាពីរូបភាពពិត ប៉ុន្តែកម្រិតជាការជ្រើសរើស៖ ត្រូវផ្ទៀងផ្ទាត់ដោយចំណុច (មេរៀនទី៦)។


វិធីសាមញ្ញបំផុតគឺប្រៀបធៀប composite រដូវប្រាំងពីរឆ្នាំ៖

```javascript
var aoi = ee.FeatureCollection('FAO/GAUL/2015/level1').filter(ee.Filter.eq('ADM1_NAME', 'Phnom Penh'));
function dry(y) {
  return ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED').filterBounds(aoi)
    .filterDate(y + '-01-01', y + '-04-30').filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 20))
    .median().divide(10000).clip(aoi);
}
function idx(img) {
  return img.normalizedDifference(['B8', 'B4']).rename('NDVI')
    .addBands(img.normalizedDifference(['B11', 'B8']).rename('NDBI'));
}
var a = idx(dry(2017)), b = idx(dry(2024));
var newBuilt = b.select('NDVI').subtract(a.select('NDVI')).lt(-0.2)
  .and(b.select('NDVI').lt(0.3))
  .and(b.select('NDBI').subtract(a.select('NDBI')).gt(0.05));
Map.addLayer(newBuilt.selfMask(), {palette: ['red']}, 'សំណង់/ដីទទេថ្មី ២០១៧–២០២៤');
```

NDBI នៅកម្ពុជាមិនបែងចែកដីទទេ ពីអគារបានល្អទេ (ដីស្ងួត និងដំបូលដែកមាន SWIR ខ្ពស់ដូចគ្នា)។ ដូច្នេះលទ្ធផលគឺ «ផ្ទៃដែលក្លាយជាមិនមែនរុក្ខជាតិ» ដែលរួមទាំងដីចាក់បំពេញ។ សម្រាប់ផែនទីទីក្រុងដែលត្រឹមត្រូវជាង ត្រូវចាត់ថ្នាក់ពីរឆ្នាំ (មេរៀនទី៦) ឬចាត់ថ្នាក់ការផ្លាស់ប្ដូរដោយផ្ទាល់ ដូចលំហាត់ក្នុងសៀវភៅចាស់របស់វគ្គនេះ ដែលប្រើចំណុច «ផ្លាស់ប្ដូរ» និង «មិនផ្លាស់ប្ដូរ» លើ composite ពីរឆ្នាំដែលដាក់ជាន់គ្នា។

!!! example "ពិសោធន៍៖ ការពង្រីកក្រុងព្រះសីហនុ ២០១៥–២០២១"
    អូសរបារដើម្បីប្រៀបធៀបរូបភាពពីរឆ្នាំ ហើយប្ដូរកម្រិត ΔNDVI និង NDBI ដើម្បីរកសំណង់ថ្មី។ រូបភាពជា Sentinel-2 ពិត ហើយផ្ទៃដែលបង្ហាញត្រូវបានគណនាពីក្រឡាពិត។

!!! note "អានរូបនេះ"

    - Dynamic World និង WorldCover មានក្រឡា ១០ ម សម្រាប់ព័ត៌មានលម្អិតថ្មីៗ។
    - ពន្លឺពេលយប់ វាស់សកម្មភាព និងអគ្គិសនី មិនមែនអគារទេ៖ ប្រើជាមួយ ដើម្បីបំពេញគ្នា។


ជាញឹកញាប់មិនចាំបាច់ចាប់ផ្ដើមពីសូន្យទេ៖

```javascript
// GHSL៖ ម² សំណង់ក្នុងក្រឡា ១០០ ម · រូបភាពមួយក្នុងមួយយុគ (១៩៧៥ … ២០៣០)
var b2000 = ee.Image('JRC/GHSL/P2023A/GHS_BUILT_S/2000').select('built_surface');
var b2020 = ee.Image('JRC/GHSL/P2023A/GHS_BUILT_S/2020').select('built_surface');
Map.addLayer(b2020.subtract(b2000).clip(aoi), {min: 0, max: 5000, palette: ['white', 'orange', 'red']}, 'GHSL សំណង់ថ្មី ២០០០–២០២០');

var dw = ee.ImageCollection('GOOGLE/DYNAMICWORLD/V1').filterBounds(aoi);
var built2024 = dw.filterDate('2024-01-01', '2025-01-01').select('built').mean().gt(0.5);   // ប្រូបាប៊ីលីតេ
var lights = ee.ImageCollection('NOAA/VIIRS/DNB/MONTHLY_V1/VCMSLCFG').filterDate('2024-01-01', '2025-01-01')
  .select('avg_rad').median();
```

ចំណុចត្រូវចាំ៖ **GHSL** មានស៊េរីវែង (១៩៧៥–២០៣០ រៀងរាល់ ៥ ឆ្នាំ) ប៉ុន្តែឆ្នាំក្រោយ ២០២០ ជាការព្យាករ ហើយក្រឡា ១០០ ម។ **Dynamic World** ផ្ដល់ប្រូបាប៊ីលីតេ «built» សម្រាប់រូបភាព Sentinel-2 នីមួយៗ នៅ ១០ ម៉ែត្រ ដែលល្អសម្រាប់ការប្ដូរថ្មីៗ។ **VIIRS** វាស់ពន្លឺពេលយប់ ដែលទាក់ទងនឹងសកម្មភាពសេដ្ឋកិច្ច និងអគ្គិសនី មិនមែនផ្ទៃអគារទេ។ ទិន្នន័យនីមួយៗមាននិយមន័យ «ទីក្រុង» ផ្ទាល់ខ្លួន ដូច្នេះលេខរបស់វាមិនស្មើគ្នាទេ។

### ១២.៣.៤. សីតុណ្ហភាពផ្ទៃដីពី Landsat

<figure markdown>
--8<-- "assets/svg/agv/a12-lst.svg"
<figcaption>រូបទី១២.៤៖ LST រួចជាស្រេចក្នុង Collection 2។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - Landsat Collection 2 Level-2 មាន ST_B10 ដែលបានកែបរិយាកាស និង emissivity រួច៖ យើងគ្រាន់តែបម្លែងឯកតា។
    - ត្រូវ mask ពពកជានិច្ច៖ ពពកត្រជាក់ខ្លាំង ហើយធ្វើឲ្យ LST ទាបខុសប្រក្រតី។
    - LST ជាសីតុណ្ហភាពផ្ទៃ (ដំបូល ផ្លូវ ស្លឹក) មិនមែនសីតុណ្ហភាពខ្យល់ដែលមនុស្សមានអារម្មណ៍ទេ។


Landsat 8 និង 9 មានឧបករណ៍ TIRS ដែលវាស់វិទ្យុសកម្មកម្ដៅ។ ក្នុង **Collection 2 Level-2** ក្រុមរលក **ST_B10** ត្រូវបានកែបរិយាកាស និង emissivity រួចជាស្រេច៖ យើងគ្រាន់តែបម្លែងឯកតា៖

```javascript
function prepL(img) {
  var qa = img.select('QA_PIXEL');
  var clear = qa.bitwiseAnd(1 << 1).eq(0)           // dilated cloud
    .and(qa.bitwiseAnd(1 << 3).eq(0))               // cloud
    .and(qa.bitwiseAnd(1 << 4).eq(0));              // cloud shadow
  var lst = img.select('ST_B10').multiply(0.00341802).add(149.0).subtract(273.15).rename('LST');
  var sr = img.select(['SR_B4', 'SR_B5']).multiply(0.0000275).add(-0.2);
  var ndvi = sr.normalizedDifference(['SR_B5', 'SR_B4']).rename('NDVI');
  return lst.addBands(ndvi).updateMask(clear).copyProperties(img, ['system:time_start']);
}
var land = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2').merge(ee.ImageCollection('LANDSAT/LC09/C02/T1_L2'))
  .filterBounds(aoi).filter(ee.Filter.calendarRange(3, 5, 'month'))       // រដូវក្ដៅ មីនា–ឧសភា
  .filterDate('2021-01-01', '2025-01-01').map(prepL);
var lst = land.select('LST').median().clip(aoi);
Map.addLayer(lst, {min: 28, max: 45, palette: ['#313695', '#74add1', '#ffffbf', '#f46d43', '#a50026']}, 'LST (°C)');
```

ចំណុចសំខាន់៖

- **LST ≠ សីតុណ្ហភាពខ្យល់**៖ LST គឺសីតុណ្ហភាពផ្ទៃ (ដំបូល ផ្លូវ ស្លឹក) ដែលអាចក្ដៅជាងខ្យល់ ១០–២០ °C នៅថ្ងៃត្រង់។
- **ពេលថត**៖ Landsat ថតប្រហែលម៉ោង ១០:៣០ ព្រឹក ដូច្នេះវាមិនបង្ហាញកំដៅអតិបរមាពេលរសៀល ឬកំដៅពេលយប់ទេ។ MODIS (MOD11A1 · MYD11A1) ផ្ដល់ LST ពេលយប់ នៅ ១ គម។
- **ក្រឡា**៖ ST_B10 ត្រូវបានវាស់នៅ ១០០ ម ហើយ resample ទៅ ៣០ ម៖ ព័ត៌មានលម្អិតតូចជាង ១០០ ម មិនពិតទេ។
- **Median ច្រើនឆ្នាំ** នៃរដូវតែមួយ កាត់បន្ថយឥទ្ធិពលថ្ងៃតែមួយ (ខ្យល់ ពពកស្ដើង)។

### ១២.៣.៥. LST និងរុក្ខជាតិ

<figure markdown>
--8<-- "assets/svg/agv/a12-lst-ndvi.svg"
<figcaption>រូបទី១២.៥៖ រុក្ខជាតិធ្វើឲ្យត្រជាក់។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - តំបន់សាងសង់ (ក្រហម) និងដីទទេក្ដៅបំផុត ទឹក (ខៀវ) ត្រជាក់បំផុត។
    - ក្នុងចំណោមដី LST ថយនៅពេល NDVI កើន៖ ការបញ្ចេញចំហាយពីស្លឹកធ្វើឲ្យផ្ទៃត្រជាក់។
    - LST ក្នុងរូបនេះត្រូវបានក្លែងធ្វើពីគម្របដីពិត ដើម្បីបង្ហាញទំនាក់ទំនង៖ លំហាត់ទី១២ ប្រើ ST_B10 ពិត។


ក្នុងទីក្រុង LST ទាក់ទងអវិជ្ជមានខ្លាំងជាមួយ NDVI៖ រុក្ខជាតិបញ្ចេញចំហាយទឹក ហើយផ្ដល់ម្លប់។ ទឹកត្រជាក់បំផុត ហើយដំបូលដែក ផ្លូវកៅស៊ូ និងដីទទេក្ដៅបំផុត។ ទំនាក់ទំនងនេះអាចវាស់ដោយ reducer៖

```javascript
var fit = land.select(['NDVI', 'LST']).median().clip(aoi).reduceRegion({
  reducer: ee.Reducer.linearFit(), geometry: aoi.geometry(), scale: 30, maxPixels: 1e9, tileScale: 4});
print('ជម្រាល °C ក្នុង NDVI ១ ឯកតា', fit.get('scale'));
```

### ១២.៣.៦. កោះកំដៅទីក្រុង

<figure markdown>
--8<-- "assets/svg/agv/a12-uhi.svg"
<figcaption>រូបទី១២.៦៖ ទីក្រុងក្ដៅជាងជនបទជុំវិញ។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - UHI = LST មធ្យមក្នុងទីក្រុង − LST មធ្យមក្នុងជនបទជុំវិញ។
    - ទន្លេ បឹង និងឧទ្យាន បង្កើត «កោះត្រជាក់» ក្នុងទីក្រុង។
    - តម្លៃ UHI អាស្រ័យលើរបៀបកំណត់ «ជនបទ»៖ ស្រែស្ងួតក្នុងរដូវប្រាំងក៏ក្ដៅដែរ។ ផ្នែកកាត់នេះជាការក្លែងធ្វើ។


**UHI (Urban Heat Island)** = LST មធ្យមក្នុងទីក្រុង − LST មធ្យមក្នុងជនបទជុំវិញ។ របៀបកំណត់ «ទីក្រុង» និង «ជនបទ» ប្ដូរលទ្ធផលច្រើន៖

```javascript
var wc = ee.ImageCollection('ESA/WorldCover/v200').first();
var urban = wc.eq(50);
var ring = aoi.geometry().buffer(10000).difference(aoi.geometry());       // ទ្រនាប់ ១០ គម
var rural = wc.eq(10).or(wc.eq(30)).or(wc.eq(40));                        // ព្រៃ ស្មៅ ដំណាំ
var lstAll = land.select('LST').median();
var tU = lstAll.updateMask(urban).reduceRegion({reducer: ee.Reducer.mean(), geometry: aoi.geometry(), scale: 30, maxPixels: 1e9});
var tR = lstAll.updateMask(rural).reduceRegion({reducer: ee.Reducer.mean(), geometry: ring, scale: 30, maxPixels: 1e9});
print('UHI (°C)', ee.Number(tU.get('LST')).subtract(tR.get('LST')));
```

ក្នុងរដូវក្ដៅ ស្រែស្ងួតជុំវិញទីក្រុងក៏ក្ដៅខ្លាំងដែរ ដូច្នេះ UHI ពេលថ្ងៃអាចតូច ឬសូម្បីអវិជ្ជមាន។ ការជ្រើសជនបទដែលមានរុក្ខជាតិ ឬការប្រើ LST ពេលយប់ (MODIS) ផ្ដល់រូបភាពពេញលេញជាង។

### ១២.៣.៧. ពីកំដៅ ទៅការប៉ះពាល់

<figure markdown>
--8<-- "assets/svg/agv/a12-workflow.svg"
<figcaption>រូបទី១២.៧៖ កំដៅ + មនុស្ស = ហានិភ័យ។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - LST តែមួយប្រាប់ថាកន្លែងណាក្ដៅ។ ការឆ្លងជាមួយប្រជាជន សាលា ឬមន្ទីរពេទ្យ ប្រាប់ថាកន្លែងណាសំខាន់។
    - Composite median រដូវក្ដៅ (មីនា–ឧសភា) ច្រើនឆ្នាំ កាត់បន្ថយឥទ្ធិពលថ្ងៃតែមួយ។
    - លទ្ធផលជួយកំណត់អាទិភាពសម្រាប់ដាំដើមឈើ ដំបូលពណ៌ស ឬម្លប់។


កំដៅក្លាយជាហានិភ័យ នៅពេលមានមនុស្សរងផលប៉ះពាល់៖

```javascript
var khan = ee.FeatureCollection('FAO/GAUL/2015/level2').filter(ee.Filter.eq('ADM1_NAME', 'Phnom Penh'));
var pop = ee.ImageCollection('WorldPop/GP/100m/pop').filter(ee.Filter.eq('country', 'KHM'))
  .filter(ee.Filter.eq('year', 2020)).first().rename('pop');
var table = lst.addBands(land.select('NDVI').median().rename('NDVI_med'))
  .reduceRegions({collection: khan, reducer: ee.Reducer.mean(), scale: 30, tileScale: 4})
  .map(function(f) {
    var p = pop.reduceRegion({reducer: ee.Reducer.sum(), geometry: f.geometry(), scale: 100, maxPixels: 1e9}).get('pop');
    return f.set('pop', p);
  });
print(table.select(['ADM2_NAME', 'LST', 'NDVI_med', 'pop']).sort('LST', false));
```

ចំណាំ៖ reduceRegions ដែលមាន reducer តែមួយ (mean) លើរូបភាពពីរក្រុមរលក ផ្ដល់ឈ្មោះលក្ខណៈតាមក្រុមរលក (`LST` និង `NDVI_med`)។ ពិនិត្យដោយ `print(table.first())` មុន select។ ព្រំដែនខណ្ឌ GAUL ២០១៥ ចាស់ជាងការបែងចែកខណ្ឌថ្មី៖ ប្រើព្រំដែនផ្លូវការថ្មីបំផុតប្រសិនបើមាន។

## ១២.៤. ឧទាហរណ៍ដែលបានដោះស្រាយ៖ ភ្នំពេញ ២០១៧–២០២៤

### ជំហានទី១៖ ការពង្រីក

គណនា `newBuilt` (ផ្នែក ១២.៣.២) និងប្រូបាប៊ីលីតេ built របស់ Dynamic World ឆ្នាំ ២០១៧ និង ២០២៤ (ផ្នែក ១២.៣.៣)។ គណនាផ្ទៃ built ក្នុងឆ្នាំនីមួយៗ ដោយ pixelArea។ ប្រៀបធៀបផ្ទៃកើនឡើងពីវិធីទាំងពីរ ហើយពន្យល់ភាពខុសគ្នា។

### ជំហានទី២៖ LST រដូវក្ដៅ

គណនា `lst` median មីនា–ឧសភា ២០២១–២០២៤ (ផ្នែក ១២.៣.៤)។ ពិនិត្យចំនួនរូបភាពស្អាតក្នុងក្រឡានីមួយៗ ដោយ `land.select('LST').count()`។

### ជំហានទី៣៖ តំបន់ពង្រីកថ្មី ក្ដៅជាងទេ?

```javascript
var stable = urban.and(newBuilt.not());                      // ទីក្រុងចាស់
var still = rural;                                           // នៅតែជាជនបទ
function meanLST(mask) {
  return lst.updateMask(mask).reduceRegion({reducer: ee.Reducer.mean(), geometry: aoi.geometry(), scale: 30, maxPixels: 1e9}).get('LST');
}
print('ថ្មី', meanLST(newBuilt), 'ចាស់', meanLST(stable), 'ជនបទ', meanLST(still));
```

### ជំហានទី៤៖ តារាងខណ្ឌ

គណនាតារាងខណ្ឌ (ផ្នែក ១២.៣.៧) ហើយនាំចេញជា CSV។ តម្រៀបតាម LST និងប្រជាជន ដើម្បីកំណត់ខណ្ឌអាទិភាព។

## ១២.៥. សកម្មភាពអនុវត្ត

- **លំហាត់ទី១២**៖ គណនា LST រដូវក្ដៅនៃភ្នំពេញ វាស់ UHI និងទំនាក់ទំនង LST–NDVI ធ្វើតារាងខណ្ឌជាមួយប្រជាជន ហើយប្រៀបធៀបការពង្រីកពី Dynamic World និង GHSL។ មើល [លំហាត់ទី១២](../workbook/lab-12.md)។
- **ក្នុងថ្នាក់**៖ ក្រុមនីមួយៗប្រើនិយមន័យ «ជនបទ» ខុសគ្នា (ស្រែ · ព្រៃ · ទ្រនាប់ ៥ ឬ ២០ គម) ហើយប្រៀបធៀបតម្លៃ UHI។

## ១២.៦. ការយល់ច្រឡំដែលត្រូវប្រុងប្រយ័ត្ន

**១. «LST គឺសីតុណ្ហភាពដែលយើងមានអារម្មណ៍»**
LST គឺសីតុណ្ហភាពផ្ទៃ។ ខ្យល់ក្នុងម្លប់អាចត្រជាក់ជាងផ្ទៃផ្លូវ ១០–២០ °C។

**២. «ST_B10 ជា DN ត្រូវកែបរិយាកាសខ្លួនឯង»**
ក្នុង Collection 2 Level-2 វាត្រូវបានកែរួច។ គ្រាន់តែគុណ ០,០០៣៤១៨០២ បូក ១៤៩ ហើយដក ២៧៣,១៥។

**៣. «NDBI ខ្ពស់ = អគារ»**
នៅកម្ពុជា ដីទទេស្ងួត និងដីចាក់បំពេញ មាន NDBI ខ្ពស់ដូចអគារ។

**៤. «UHI ជាតម្លៃថេរនៃទីក្រុងមួយ»**
វាប្ដូរតាមរដូវ ពេលថ្ងៃ/យប់ និងនិយមន័យជនបទ។ ត្រូវរាយការណ៍វិធីគណនា។

**៥. «ទិន្នន័យទីក្រុងសកលទាំងអស់គួរតែផ្ដល់លេខដូចគ្នា»**
និយមន័យ ក្រឡា និងឆ្នាំខុសគ្នា។ ជ្រើសទិន្នន័យតាមសំណួរ ហើយប្រើតែមួយពេលប្រៀបធៀបឆ្នាំ។

## ១២.៧. សេចក្ដីសង្ខេបមេរៀន

ការពង្រីកទីក្រុងបង្ហាញជារុក្ខជាតិបាត់ ដីទទេ និងសំណង់ថ្មី។ ការប្ដូរ NDVI និង NDBI រវាងពីរឆ្នាំ រកផ្ទៃដែលក្លាយជាមិនមែនរុក្ខជាតិ ប៉ុន្តែមិនបែងចែកដីចាក់បំពេញ ពីអគារបានល្អទេ។ GHSL Dynamic World WorldCover និង VIIRS ផ្ដល់ការវាស់ទីក្រុងរួចជាស្រេច ដោយនិយមន័យ និងក្រឡាខុសគ្នា។

LST ពី Landsat C2 L2 ST_B10 គណនាដោយការបម្លែងឯកតាសាមញ្ញ បន្ទាប់ពីលុបពពក។ LST ទាក់ទងអវិជ្ជមានជាមួយ NDVI ហើយ UHI អាស្រ័យលើនិយមន័យទីក្រុង និងជនបទ។ ការភ្ជាប់ LST ជាមួយប្រជាជនតាមខណ្ឌ ជួយកំណត់តំបន់អាទិភាពសម្រាប់ដើមឈើ និងម្លប់។

!!! abstract "គំនិតស្នូលនៃមេរៀន"
    ទីក្រុងប្ដូរគម្របដី ហើយគម្របដីប្ដូរសីតុណ្ហភាព។ ផែនទីកំដៅមានប្រយោជន៍បំផុត នៅពេលវាប្រាប់ពីកន្លែងដែលកំដៅ និងមនុស្សជួបគ្នា ជាមួយការពន្យល់ច្បាស់ថា LST វាស់អ្វី និងមិនវាស់អ្វី។
