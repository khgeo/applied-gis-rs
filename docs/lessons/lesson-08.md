# មេរៀនទី៨៖ គ្រោះរាំងស្ងួត ទឹកភ្លៀង និងសុខភាពរុក្ខជាតិ

គ្រោះរាំងស្ងួតខុសពីទឹកជំនន់៖ វាមកយឺតៗ គ្មានព្រំដែនច្បាស់ ហើយជាញឹកញាប់ត្រូវបានគេដឹងតែនៅពេលស្រូវស្លាប់រួចទៅហើយ។ នៅកម្ពុជា ឆ្នាំ El Niño ដូចជា ២០១៥–២០១៦ និង ២០១៩ បាននាំមកនូវរដូវវស្សាចាប់ផ្ដើមយឺត រយៈពេលស្ងួតក្នុងរដូវ (dry spell) និងការខូចខាតដំណាំនៅខេត្តជាច្រើន។

គ្រោះរាំងស្ងួតមិនមែនជា «ភ្លៀងតិច» ទេ ប៉ុន្តែជា **«ភ្លៀងតិចជាងធម្មតា សម្រាប់ទីកន្លែង និងពេលនោះ»**។ ១០០ មីលីម៉ែត្រក្នុងខែកក្កដា ជាគ្រោះរាំងស្ងួតនៅខេត្តកោះកុង ប៉ុន្តែជាភ្លៀងល្អនៅខេត្តបាត់ដំបង។ មេរៀននេះបង្រៀនរបៀបវាស់ «ធម្មតា» និង «ខុសពីធម្មតា» ដោយទិន្នន័យទឹកភ្លៀង CHIRPS និងសន្ទស្សន៍រុក្ខជាតិ MODIS ក្នុង Earth Engine។

## ៨.១. គោលបំណងសិក្សា

បន្ទាប់ពីបញ្ចប់មេរៀន និស្សិតអាច៖

1. បែងចែកគ្រោះរាំងស្ងួតបួនប្រភេទ ហើយផ្គូផ្គងសន្ទស្សន៍ផ្កាយរណបទៅប្រភេទនីមួយៗ។
2. បង្កើតទឹកភ្លៀងប្រចាំខែ និង climatology ពី CHIRPS ហើយគណនា anomaly ជា % និង z-score។
3. ពន្យល់ SPI និងរយៈពេលបូក (១ ៣ ៦ ខែ) ហើយគណនា SPI សាមញ្ញក្នុង Earth Engine។
4. គណនា VCI TCI និង VHI ពី MODIS ហើយបកស្រាយកម្រិតគ្រោះរាំងស្ងួត។
5. រួមបញ្ចូលសន្ទស្សន៍ទឹកភ្លៀង និងរុក្ខជាតិ ដើម្បីវាយតម្លៃគ្រោះរាំងស្ងួតតាមខេត្ត ជាមួយដែនកំណត់ច្បាស់លាស់។

## ៨.២. ស្ថានភាពបើកមេរៀន៖ រដូវវស្សា ២០១៩

ក្នុងខែកក្កដា ២០១៩ កសិករនៅខេត្តកំពង់ស្ពឺ តាកែវ និងខេត្តជាច្រើនទៀត រាយការណ៍ថាភ្លៀងមកយឺត ហើយស្រូវដែលទើបស្ទូងចាប់ផ្ដើមស្រពោន។ ក្រសួងកសិកម្មត្រូវការដឹងយ៉ាងឆាប់ថា **ខេត្ត និងស្រុកណាខ្លះរងផលប៉ះពាល់ខ្លាំងជាងគេ** ដើម្បីរៀបចំការផ្គត់ផ្គង់ទឹក និងពូជ។ ស្ថានីយវាស់ទឹកភ្លៀងមានតិច ហើយការស្ទង់វាលត្រូវការពេលច្រើនសប្ដាហ៍។

សំណួរដែលមេរៀននេះឆ្លើយ៖

- តើត្រូវកំណត់ «ភ្លៀងធម្មតា» សម្រាប់ទីកន្លែង និងខែនីមួយៗយ៉ាងដូចម្ដេច?
- តើ «ខ្វះប៉ុន្មាន» គួរវាស់ជា % ឬជា z-score?
- តើរុក្ខជាតិពិតជារងផលប៉ះពាល់ឬទេ ហើយត្រូវវាស់វាដោយរបៀបណា?
- តើសន្ទស្សន៍ទាំងនេះមានដែនកំណត់អ្វីខ្លះ?

ប្រភេទទាំងនេះកើតជាដំណាក់កាល។ រយៈពេលស្ងួតខ្លី អាចជាគ្រោះរាំងស្ងួតឧតុនិយម ដោយមិនប៉ះពាល់ដំណាំទេ ប្រសិនបើដីនៅមានសំណើម ឬស្រែមានប្រព័ន្ធធារាសាស្ត្រ។ ដូច្នេះសន្ទស្សន៍ទឹកភ្លៀងតែមួយ មិនគ្រប់គ្រាន់ដើម្បីនិយាយថា «ដំណាំរងគ្រោះ» ទេ។

### ៨.៣.២. ទឹកភ្លៀង CHIRPS

**CHIRPS** (Climate Hazards Group InfraRed Precipitation with Station data) រួមបញ្ចូលរូបភាពផ្កាយរណបអ៊ីនហ្វ្រារ៉េដ ជាមួយទិន្នន័យស្ថានីយ ដើម្បីផ្ដល់ទឹកភ្លៀង **ប្រចាំថ្ងៃ ចាប់ពីឆ្នាំ ១៩៨១** នៅក្រឡា ~៥,៥ គីឡូម៉ែត្រ។ ប្រវត្តិយូរនេះសំខាន់ណាស់ ព្រោះការគណនា «ធម្មតា» ត្រូវការយ៉ាងហោចណាស់ ៣០ ឆ្នាំ។

```javascript
var chirps = ee.ImageCollection('UCSB-CHG/CHIRPS/DAILY').select('precipitation');
var kh = ee.FeatureCollection('FAO/GAUL/2015/level0').filter(ee.Filter.eq('ADM0_NAME', 'Cambodia'));

// ទឹកភ្លៀងសរុបប្រចាំខែ (mm)៖ sum() មិនមែន mean()
function monthly(y, m) {
  var start = ee.Date.fromYMD(y, m, 1);
  return chirps.filterDate(start, start.advance(1, 'month')).sum()
    .set({year: y, month: m, 'system:time_start': start.millis()});
}
var jul2019 = monthly(2019, 7);
Map.addLayer(jul2019.clip(kh), {min: 0, max: 400, palette: ['#fff5eb', '#6baed6', '#08306b']}, 'ភ្លៀង កក្កដា ២០១៩');
```

ចំណាំ៖ សម្រាប់ទឹកភ្លៀង យើងប្រើ **sum()** ដើម្បីបានសរុបក្នុងខែ។ ការប្រើ mean() ផ្ដល់ «mm ក្នុងមួយថ្ងៃ» ដែលជាឯកតាផ្សេង។ CHIRPS ក៏មាន collection `UCSB-CHG/CHIRPS/PENTAD` (៥ ថ្ងៃ) ដែលលឿនជាងសម្រាប់រយៈពេលវែង។

### ៨.៣.៣. Climatology និង anomaly

<figure markdown>
--8<-- "assets/svg/agv/a08-climatology.svg"
<figcaption>រូបទី៨.២៖ «ធម្មតា» គឺជាចន្លោះ មិនមែនលេខមួយទេ។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - Climatology = ស្ថិតិនៃខែដូចគ្នាក្នុងរយៈពេលវែង (ជាទូទៅ ៣០ ឆ្នាំ)។
    - ខែមួយប្រៀបធៀបតែនឹងខែដដែលនៃឆ្នាំផ្សេងៗប៉ុណ្ណោះ៖ កក្កដា ២០១៩ ធៀបនឹងកក្កដាទាំងអស់។
    - តម្លៃមធ្យមប្រចាំខែ ផ្អែកលើកម្រិតប្រហាក់ប្រហែលនៃភ្នំពេញ ស៊េរីឆ្នាំជាស៊េរីក្លែងធ្វើ។


**Climatology** គឺជាស្ថិតិនៃខែដូចគ្នា ក្នុងរយៈពេលយោងវែង (ជាទូទៅ ១៩៩១–២០២០)។ ដើម្បីដឹងថាខែកក្កដា ២០១៩ ស្ងួតប៉ុណ្ណា យើងប្រៀបធៀបវាជាមួយខែកក្កដាទាំង ៣០ ក្នុងរយៈពេលយោង៖

```javascript
var years = ee.List.sequence(1991, 2020);
var julys = ee.ImageCollection.fromImages(years.map(function(y) { return monthly(ee.Number(y), 7); }));
var mean = julys.mean(), sd = julys.reduce(ee.Reducer.stdDev());
var pctNormal = jul2019.divide(mean).multiply(100).rename('pct');
var z = jul2019.subtract(mean).divide(sd).rename('z');
Map.addLayer(pctNormal.clip(kh), {min: 40, max: 160, palette: ['#8c510a', '#f6e8c3', '#5ab4ac', '#01665e']}, '% នៃធម្មតា');
Map.addLayer(z.clip(kh), {min: -2, max: 2, palette: ['#b2182b', '#f7f7f7', '#2166ac']}, 'z-score');
```

!!! note "អានរូបនេះ"
    - % នៃធម្មតាងាយយល់ ប៉ុន្តែខែប្រាំង (ភ្លៀង ១០ mm) ធ្វើឲ្យភាគរយប្រែប្រួលខ្លាំងគ្មានន័យ។
    - z-score គិតពីភាពប្រែប្រួលធម្មតា៖ −១ ដល់ −២ = ស្ងួតជាងធម្មតាច្បាស់។
    - SPI ពិតប្រើការបម្លែង gamma មុនគណនា z ព្រោះទឹកភ្លៀងមិនចែកចាយតាមរាង normal។


**% នៃធម្មតា** ងាយពន្យល់ដល់អ្នកសម្រេចចិត្ត ប៉ុន្តែវាបំភាន់ក្នុងខែប្រាំង៖ ភ្លៀង ២០ mm ធៀបនឹងមធ្យម ១០ mm គឺ ២០០%។ **z-score** គិតពីភាពប្រែប្រួលធម្មតានៃខែនោះ៖ z = −១ មានន័យថាស្ងួតជាងធម្មតាមួយគម្លាតស្តង់ដារ ដែលកើតប្រហែល ១ ឆ្នាំក្នុង ៦ ឆ្នាំ។

### ៨.៣.៤. SPI និងរយៈពេលបូក

<figure markdown>
--8<-- "assets/svg/agv/a08-spi.svg"
<figcaption>រូបទី៨.៤៖ រយៈពេលខ្លី ឬវែង ឆ្លើយសំណួរខុសគ្នា។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - SPI-១ លោតឡើងចុះរាល់ខែ៖ ល្អសម្រាប់ដឹងភ្លាមៗ តែមានសំឡេងរំខានច្រើន។
    - SPI-៣ ទាក់ទងនឹងសំណើមដី និងដំណាំ។ SPI-៦ ដល់ ១២ ទាក់ទងនឹងទន្លេ និងអាងស្តុកទឹក។
    - SPI < −១ = ស្ងួតមធ្យម · < −១,៥ = ស្ងួតខ្លាំង · < −២ = ស្ងួតខ្លាំងបំផុត។


**SPI (Standardized Precipitation Index)** គឺជា z-score នៃទឹកភ្លៀង **បូកក្នុងរយៈពេល k ខែ** (SPI-១ SPI-៣ SPI-៦ SPI-១២)។ ក្នុងការគណនាផ្លូវការ ទឹកភ្លៀងត្រូវបានបម្លែងតាមការចែកចាយ gamma មុនគណនា z ព្រោះទឹកភ្លៀងមិនចែកចាយតាមរាង normal ទេ (មិនអាចអវិជ្ជមាន និងលំអៀងទៅខាងស្ដាំ)។ ក្នុង Earth Engine យើងអាចគណនា **SPI សាមញ្ញ** (z-score នៃទឹកភ្លៀងបូក) ដែលល្អគ្រប់គ្រាន់សម្រាប់ការបង្រៀន និងការមើលលំនាំលំហ៖

```javascript
// SPI-3 សាមញ្ញ សម្រាប់ខែ ឧសភា–កក្កដា
function acc3(y) {
  var start = ee.Date.fromYMD(y, 5, 1);
  return chirps.filterDate(start, start.advance(3, 'month')).sum();
}
var ref = ee.ImageCollection.fromImages(years.map(function(y) { return acc3(ee.Number(y)); }));
var spi3 = acc3(2019).subtract(ref.mean()).divide(ref.reduce(ee.Reducer.stdDev())).rename('SPI3');
```

|---|---|
| ≥ ២,០ | សើមខ្លាំងបំផុត |
| ១,០ ដល់ ២,០ | សើម |
| −១,០ ដល់ ១,០ | ជិតធម្មតា |
| −១,៥ ដល់ −១,០ | ស្ងួតមធ្យម |
| −២,០ ដល់ −១,៥ | ស្ងួតខ្លាំង |
| ≤ −២,០ | ស្ងួតខ្លាំងបំផុត |

**SPI-១** ប្រតិកម្មលឿន តែមានសំឡេងរំខាន។ **SPI-៣** ទាក់ទងនឹងសំណើមដី និងដំណាំរដូវវស្សា។ **SPI-៦ ដល់ ១២** ទាក់ទងនឹងលំហូរទន្លេ អាងស្តុកទឹក និងទឹកក្រោមដី។

!!! example "ពិសោធន៍៖ SPI តាមរយៈពេលបូក"

### ៨.៣.៥. VCI៖ ស្ថានភាពរុក្ខជាតិ

<figure markdown>
--8<-- "assets/svg/agv/a08-vci.svg"
<figcaption>រូបទី៨.៥៖ ធៀបនឹងខ្លួនឯង មិនមែននឹងកន្លែងផ្សេង។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - VCI ដាក់ NDVI បច្ចុប្បន្នក្នុងចន្លោះ min–max នៃក្រឡាដដែល ក្នុងពេលដដែលនៃឆ្នាំ។
    - ដូច្នេះព្រៃ (NDVI ខ្ពស់) និងស្រែ (NDVI ទាប) អាចប្រៀបធៀបគ្នាបាន៖ ០ = អាក្រក់បំផុតដែលធ្លាប់មាន។
    - ត្រូវការប្រវត្តិយូរ (MODIS ២០០០–បច្ចុប្បន្ន) ដើម្បីឲ្យ min និង max មានន័យ។


**VCI (Vegetation Condition Index)** ដែលស្នើដោយ Kogan ដាក់ NDVI បច្ចុប្បន្ន ក្នុងចន្លោះ min–max នៃក្រឡាដដែល ក្នុងពេលដដែលនៃឆ្នាំ៖

VCI = (NDVI − NDVImin) / (NDVImax − NDVImin) × ១០០

VCI = ០ មានន័យថា NDVI ទាបបំផុតដែលធ្លាប់ឃើញសម្រាប់ខែនេះ នៅក្រឡានេះ។ ការធ្វើឲ្យស្តង់ដារនេះ អនុញ្ញាតឲ្យប្រៀបធៀបព្រៃ ស្រែ និងវាលស្មៅ ដែលមាន NDVI ខុសគ្នាធម្មជាតិ។ យើងប្រើ **MODIS MOD13Q1** (NDVI ១៦ ថ្ងៃ ២៥០ ម ចាប់ពីឆ្នាំ ២០០០) ព្រោះវាមានប្រវត្តិយូរ៖

```javascript
var mod = ee.ImageCollection('MODIS/061/MOD13Q1').select('NDVI');
function ndviMonth(col, m) { return col.filter(ee.Filter.calendarRange(m, m, 'month')); }
var hist = ndviMonth(mod.filterDate('2001-01-01', '2024-01-01'), 8);         // ខែសីហា ២០០១–២០២៣
var nmin = hist.min(), nmax = hist.max();
var cur = ndviMonth(mod.filterDate('2019-01-01', '2020-01-01'), 8).mean();
var vci = cur.subtract(nmin).divide(nmax.subtract(nmin)).multiply(100).rename('VCI');
Map.addLayer(vci.clip(kh), {min: 0, max: 100, palette: ['#a50026', '#fdae61', '#ffffbf', '#a6d96a', '#006837']}, 'VCI សីហា ២០១៩');
```

MOD13Q1 NDVI មានមាត្រដ្ឋាន ០,០០០១ ប៉ុន្តែក្នុង VCI មាត្រដ្ឋាននេះលុបគ្នាចេញ ដូច្នេះមិនចាំបាច់គុណទេ។ VCI < ៤០ ជាទូទៅបង្ហាញស្ថានភាពរាំងស្ងួត ហើយ < ២០ ធ្ងន់ធ្ងរ។ ចំណាំថា **min និង max ងាយរងឥទ្ធិពលពពក**៖ NDVI min ដែលមកពីពពក ធ្វើឲ្យ VCI ខ្ពស់ពេក។ អ្នកស្រាវជ្រាវច្រើនតែប្រើ percentile ៥ និង ៩៥ ជំនួស min និង max ឬប្រើតែក្រឡាដែល SummaryQA ល្អ។

### ៨.៣.៦. TCI និង VHI

<figure markdown>
--8<-- "assets/svg/agv/a08-vhi.svg"
<figcaption>រូបទី៨.៦៖ បៃតងតិច + ក្ដៅខ្លាំង = រាំងស្ងួត។</figcaption>
</figure>

!!! note "អានរូបនេះ"
    - TCI ខ្ពស់ពេល LST ទាប (ត្រជាក់) ព្រោះរុក្ខជាតិដែលមានទឹក ត្រជាក់ខ្លួនដោយការបញ្ចេញចំហាយ។
    - VHI ទាបទាល់តែ VCI ឬ TCI ទាប៖ ក្រឡាក្រហមនៅជ្រុងខាងក្រោមឆ្វេង។
    - ពេលពពកច្រើន LST បាត់ច្រើន៖ VHI រដូវវស្សាត្រូវបកស្រាយដោយប្រុងប្រយ័ត្ន។


រុក្ខជាតិដែលខ្វះទឹក បិទរន្ធស្លឹក ហើយឈប់ត្រជាក់ខ្លួនដោយការបញ្ចេញចំហាយ៖ ផ្ទៃរបស់វាក្ដៅឡើង។ **TCI (Temperature Condition Index)** វាស់ចំណុចនេះពីសីតុណ្ហភាពផ្ទៃដី (MODIS MOD11A2 ៨ ថ្ងៃ ១ គម)៖

TCI = (LSTmax − LST) / (LSTmax − LSTmin) × ១០០

សូមកត់សម្គាល់ថាលំដាប់ត្រូវបានបញ្ច្រាស៖ **ក្ដៅ = TCI ទាប = អាក្រក់**។ **VHI (Vegetation Health Index)** រួមបញ្ចូលទាំងពីរ៖ VHI = ០,៥ × VCI + ០,៥ × TCI។

```javascript
var lst = ee.ImageCollection('MODIS/061/MOD11A2').select('LST_Day_1km');
var lhist = ndviMonth(lst.filterDate('2001-01-01', '2024-01-01'), 8);
var lmin = lhist.min(), lmax = lhist.max();
var lcur = ndviMonth(lst.filterDate('2019-01-01', '2020-01-01'), 8).mean();
var tci = lmax.subtract(lcur).divide(lmax.subtract(lmin)).multiply(100).rename('TCI');
var vhi = vci.multiply(0.5).add(tci.resample('bilinear').multiply(0.5)).rename('VHI');
```

ក្នុងរដូវវស្សា LST មានចន្លោះច្រើនដោយសារពពក (MOD11A2 ជា composite ៨ ថ្ងៃ ប៉ុន្តែនៅតែខ្វះ)។ ដូច្នេះ VHI រដូវវស្សាត្រូវបកស្រាយដោយប្រុងប្រយ័ត្ន ហើយពិនិត្យស្រទាប់ចំនួនការសង្កេតជាមួយ។

- **SPI/CHIRPS** ជាសញ្ញាព្រមាន **មុន**៖ វាប្រាប់ពីហានិភ័យ មុនពេលដំណាំខូច។
- **VCI/VHI** ជាសញ្ញា **បញ្ជាក់**៖ វាប្រាប់ថារុក្ខជាតិពិតជារងផលប៉ះពាល់។
- ប្រសិនបើ SPI ទាប ប៉ុន្តែ VCI ធម្មតា ប្រហែលជាមានប្រព័ន្ធធារាសាស្ត្រ ឬទឹកក្នុងដីនៅគ្រប់គ្រាន់។
- ប្រសិនបើ VCI ទាប ប៉ុន្តែ SPI ធម្មតា មូលហេតុអាចជាអ្វីផ្សេង៖ ទឹកជំនន់ ដំណាំដាំយឺត ជំងឺ ឬការប្ដូរដំណាំ។

### ៨.៣.៨. mask ដំណាំ និងស្ថិតិតាមខេត្ត

គ្រោះរាំងស្ងួតកសិកម្មត្រូវវាស់លើ **ដីកសិកម្ម** មិនមែនលើព្រៃ ឬទីក្រុងទេ។ ប្រើ ESA WorldCover (ថ្នាក់ ៤០) ឬផែនទីស្រែ (មេរៀនទី១១) ជា mask មុនពេលគណនាស្ថិតិតាមខេត្ត៖

```javascript
var crop = ee.ImageCollection('ESA/WorldCover/v200').first().eq(40);
var provs = ee.FeatureCollection('FAO/GAUL/2015/level1').filter(ee.Filter.eq('ADM0_NAME', 'Cambodia'));
var table = vhi.updateMask(crop.reproject(vhi.projection())).reduceRegions({
  collection: provs, reducer: ee.Reducer.mean(), scale: 250, tileScale: 4});
print(table.sort('mean').select(['ADM1_NAME', 'mean']).limit(10));
```

## ៨.៤. ឧទាហរណ៍ដែលបានដោះស្រាយ៖ រដូវវស្សា ២០១៩ តាមខេត្ត

## ៨.៥. សកម្មភាពអនុវត្ត

## ៨.៦. ការយល់ច្រឡំដែលត្រូវប្រុងប្រយ័ត្ន

**១. «ភ្លៀងតិច = គ្រោះរាំងស្ងួត»**
គ្រោះរាំងស្ងួតគឺភ្លៀងតិចជាងធម្មតា សម្រាប់ទីកន្លែង និងខែនោះ។ ខែមករាតែងតែស្ងួត ប៉ុន្តែមិនមែនគ្រោះរាំងស្ងួតទេ។

**២. «mean() នៃ CHIRPS ប្រចាំថ្ងៃ ផ្ដល់ទឹកភ្លៀងប្រចាំខែ»**
mean() ផ្ដល់ mm/ថ្ងៃ។ ទឹកភ្លៀងសរុបប្រចាំខែ ត្រូវប្រើ sum()។

**៣. «VCI ទាប មានន័យថាគ្មានភ្លៀង»**
VCI ទាប អាចមកពីទឹកជំនន់ ការដាំយឺត ការច្រូតមុន ឬពពកដែលរត់ចួល។ ត្រូវពិនិត្យជាមួយ SPI និងប្រតិទិនដំណាំ។

**៤. «រយៈពេលយោងណាក៏បាន»**
ការប្ដូររយៈពេលយោងប្ដូរ «ធម្មតា» ហើយប្ដូរ anomaly ទាំងអស់។ ប្រើរយៈពេលស្តង់ដារ (១៩៩១–២០២០) ហើយរាយការណ៍វា។

**៥. «សន្ទស្សន៍ផ្កាយរណបវាស់ការខូចខាតដំណាំ»**
វាវាស់លក្ខខណ្ឌ (ភ្លៀង ភាពបៃតង កំដៅ)។ ការខូចខាតទិន្នផលត្រូវការការស្ទង់ ឬម៉ូដែលដំណាំ។

## ៨.៧. សេចក្ដីសង្ខេបមេរៀន

គ្រោះរាំងស្ងួតរីកដាលពីឧតុនិយម ទៅកសិកម្ម ជលសាស្ត្រ និងសង្គមសេដ្ឋកិច្ច។ CHIRPS ផ្ដល់ទឹកភ្លៀងប្រចាំថ្ងៃចាប់ពីឆ្នាំ ១៩៨១ ដែលយើងបូកជាខែ ហើយប្រៀបធៀបជាមួយ climatology ជា % នៃធម្មតា z-score និង SPI តាមរយៈពេលបូក ១ ៣ ឬ ៦ ខែ។

VCI ដាក់ NDVI បច្ចុប្បន្នក្នុងចន្លោះប្រវត្តិសាស្ត្រនៃក្រឡាដដែល TCI ធ្វើដូចគ្នាជាមួយ LST ហើយ VHI រួមបញ្ចូលទាំងពីរ។ ដោយសាររុក្ខជាតិឆ្លើយតបយឺត SPI ជាសញ្ញាព្រមាន ហើយ VHI ជាការបញ្ជាក់។ ការរួមបញ្ចូលទាំងពីរ លើដីដំណាំ តាមខេត្ត ផ្ដល់ការវាយតម្លៃដែលមានប្រយោជន៍ ប៉ុន្តែត្រូវរាយការណ៍ដែនកំណត់ជានិច្ច។

!!! abstract "គំនិតស្នូលនៃមេរៀន"
    គ្រោះរាំងស្ងួតតែងតែជាការប្រៀបធៀប៖ ខែនេះ ធៀបនឹងខែដដែលក្នុងឆ្នាំផ្សេងៗ នៅកន្លែងដដែល។ ដូច្នេះប្រវត្តិទិន្នន័យវែង និងរយៈពេលយោងច្បាស់លាស់ សំខាន់ដូចរូបភាពបច្ចុប្បន្នដែរ។

10. ស្នើប្រព័ន្ធព្រមានគ្រោះរាំងស្ងួតប្រចាំខែសម្រាប់កម្ពុជា ដោយប្រើសន្ទស្សន៍ក្នុងមេរៀននេះ៖ អ្វី ពេលណា និងកម្រិតណា?

### គ. កិច្ចការខ្លី

គណនា SPI-៣ ខែឧសភា–កក្កដា សម្រាប់ឆ្នាំ ២០១៥ ២០១៩ និង ២០២៣ លើខេត្តរបស់អ្នក។ ធ្វើតារាង ហើយសរសេរកថាខណ្ឌមួយប្រៀបធៀបជាមួយព័ត៌មាន ឬការចងចាំរបស់គ្រួសារអំពីឆ្នាំទាំងនោះ។
