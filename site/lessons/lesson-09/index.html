    មេរៀនទី៩៖ ទីសណ្ឋាន ជម្រាល និងអាងទន្លេ¶

    ព័ត៌មានមេរៀន

    រយៈពេលបង្រៀនដែលស្នើ៖ ៣ ម៉ោង · មេរៀនទី ៩ ក្នុងចំណោម ១៥

    ស្លាយបង្រៀនមេរៀននេះ

    កម្ពុជាជាប្រទេសដែលភាគច្រើនរាបស្មើ៖ វាលទំនាបទន្លេមេគង្គ និងទន្លេសាប ព័ទ្ធជុំវិញដោយជួរភ្នំក្រវាញ ដំរី ដងរែក និងខ្ពង់រាបភាគឦសាន។ ភាពរាបស្មើនេះធ្វើឱ្យកម្ពស់ដីពីរបីម៉ែត្រ សំខាន់ខ្លាំងណាស់៖ វាសម្រេចថាភូមិណាលិចទឹក ស្រែណាទទួលទឹក និងទឹកភ្លៀងហូរទៅទន្លេណា។

    មេរៀននេះបង្រៀនការប្រើ គំរូកម្ពស់ឌីជីថល (DEM) ក្នុង Earth Engine៖ ជ្រើស DEM ដែលសមស្រប គណនាជម្រាល ទិស និងស្រមោលភ្នំ យល់ពីទិសលំហូរ និងស្រុតទឹក ប្រើអាងរងទឹក HydroBASINS និងកម្ពស់ពីលើប្រឡាយទឹក (HAND) សម្រាប់ការវិភាគហានិភ័យទឹកជំនន់។

    ៩.១. គោលបំណងសិក្សា¶

    បន្ទាប់ពីបញ្ចប់មេរៀន និស្សិតអាច៖

    1. ប្រៀបធៀប DEM ដែលមានក្នុង Earth Engine ហើយពន្យល់ភាពខុសគ្នារវាង DSM និង DTM។
    2. គណនាជម្រាល ទិស និងស្រមោលភ្នំ ដោយ ee.Terrain ហើយបកស្រាយលទ្ធផល។
    3. ពន្យល់ទិសលំហូរ D8 ស្រុតទឹក និងការកំណត់អាងរងទឹក ពី DEM។
    4. ប្រើ HydroBASINS HydroRIVERS និង MERIT Hydro ដើម្បីវិភាគតាមអាងទន្លេ។
    5. ប្រើ HAND ដើម្បីកំណត់តំបន់ងាយលិចទឹក ហើយគណនាស្ថិតិតាមអាងរង។

    ចំណេះដឹងមុនត្រូវមាន¶

    • មេរៀនទី៤៖ reduceRegions · មេរៀនទី៧៖ mask ជម្រាល និង HAND។
    • សៀវភៅទី២៖ ការវិភាគលើផ្ទៃ (slope aspect hillshade) ក្នុង QGIS។

    ៩.២. ស្ថានភាពបើកមេរៀន៖ អាងរងទឹកជុំវិញទន្លេសាប¶

    គម្រោងគ្រប់គ្រងធនធានទឹកមួយ ត្រូវការព័ត៌មានអំពីអាងរងទឹកនីមួយៗ ដែលហូរចូលទន្លេសាប៖ ផ្ទៃ កម្ពស់មធ្យម ជម្រាល ទឹកភ្លៀងប្រចាំឆ្នាំ និងភាគរយនៃដីដែលងាយលិចទឹក។ ពួកគេចង់ដឹងថាអាងរងណាដែលទឹកហូរមកលឿន (ជម្រាលខ្លាំង ភ្លៀងច្រើន) ហើយអាងរងណាដែលមានដីទំនាបច្រើនដែលងាយលិច។

    សំណួរដែលមេរៀននេះឆ្លើយ៖

    • តើ DEM មួយណាសមសម្រាប់វាលទំនាបកម្ពុជា ហើយវាមានកំហុសអ្វីខ្លះ?
    • តើទឹកហូរទៅណា ហើយព្រំអាងរងទឹកត្រូវកំណត់យ៉ាងដូចម្ដេច?
    • តើ «ទីទាប» គួរវាស់ធៀបនឹងនីវ៉ូសមុទ្រ ឬធៀបនឹងទន្លេជិតបំផុត?

    ៩.៣. ទ្រឹស្ដីស្នូល¶

    ៩.៣.១. DEM ក្នុង Earth Engine¶

    អានរូបនេះ

    • SRTM Copernicus និង ALOS វាស់ផ្ទៃខាងលើ (ដំបូល ពំនូកព្រៃ) មិនមែនដីទេ៖ ព្រៃខ្ពស់ធ្វើឱ្យកម្ពស់លើស។
    • សម្រាប់ជលសាស្ត្រ ប្រើ MERIT Hydro ឬ HydroSHEDS ដែលបានកែលម្អរួច។
    • វាលទំនាបកម្ពុជាមានជម្រាលតិចណាស់៖ កំហុស DEM ពីរបីម៉ែត្រ មានឥទ្ធិពលធំលើលំហូរទឹក។

    DEM ពិភពលោកភាគច្រើនក្នុង Earth Engine មានក្រឡា ៣០ ម៉ែត្រ ឬ ៩០ ម៉ែត្រ៖

    var srtm = ee.Image('USGS/SRTMGL1_003').select('elevation');
var glo30 = ee.ImageCollection('COPERNICUS/DEM/GLO30').select('DEM')
  .mosaic().setDefaultProjection('EPSG:4326', null, 30);
var merit = ee.Image('MERIT/Hydro/v1_0_1');      // elv dir upa hnd wth ...
Map.addLayer(srtm, {min: 0, max: 800, palette: ['#1a9850', '#fee08b', '#a6611a', 'white']}, 'SRTM');

    ចំណុចសំខាន់បំផុត៖ SRTM Copernicus និង ALOS ជា DSM (Digital Surface Model)៖ វាវាស់ផ្ទៃខាងលើបំផុត រួមទាំងពំនូកព្រៃ និងដំបូលផ្ទះ មិនមែនផ្ទៃដី (DTM) ទេ។ នៅតំបន់ព្រៃក្រាស់ កម្ពស់អាចលើសពីដីពិត ៥–២០ ម៉ែត្រ។ MERIT DEM និង HydroSHEDS ត្រូវបានកែលម្អ (ដកឥទ្ធិពលរុក្ខជាតិ និងសំឡេងរំខាន ហើយកែសម្រាប់លំហូរទឹក) ដូច្នេះវាសមជាងសម្រាប់ជលសាស្ត្រ ទោះបីក្រឡាធំជាងក៏ដោយ។

    នៅវាលទំនាបកម្ពុជា ភាពខុសគ្នានៃកម្ពស់រវាងកន្លែងជិតគ្នាតូចណាស់ (តិចជាង ១ ម៉ែត្រក្នុង ១ គីឡូម៉ែត្រ)។ កំហុស DEM ពីរបីម៉ែត្រ អាចធំជាងសញ្ញាពិត ដូច្នេះលទ្ធផលជលសាស្ត្រនៅតំបន់ទាំងនោះត្រូវបកស្រាយដោយប្រុងប្រយ័ត្ន។

    ៩.៣.២. ជម្រាល ទិស និងស្រមោលភ្នំ¶

    អានរូបនេះ

    • slope៖ មុំជាដឺក្រេ (០ = រាបស្មើ)។ aspect៖ ទិសដែលជម្រាលបែរទៅ (០ = ជើង ៩០ = កើត)។
    • hillshade៖ ស្រមោលក្លែងធ្វើពីពន្លឺព្រះអាទិត្យ សម្រាប់មើលរូបរាងដី (សៀវភៅទី១)។
    • គណនានៅ projection ដែលមានម៉ែត្រ៖ mosaic ត្រូវ setDefaultProjection (មេរៀនទី៦)។

    ee.Terrain គណនាផលិតផលទីសណ្ឋានពីក្រឡាជិតខាង ៣ × ៣៖

    var t = ee.Terrain.products(srtm);                  // elevation slope aspect hillshade
Map.addLayer(t.select('slope'), {min: 0, max: 30, palette: ['#ffffcc', '#fd8d3c', '#bd0026']}, 'ជម្រាល (°)');
Map.addLayer(ee.Terrain.hillshade(srtm, 315, 35), {min: 0, max: 255}, 'hillshade', false);
    ផលិតផល ឯកតា ប្រើសម្រាប់
    slope ដឺក្រេ (០–៩០) ហានិភ័យបាក់ដី សមស្របកសិកម្ម mask ទឹកជំនន់
    aspect ដឺក្រេ (០ ជើង · ៩០ កើត · ១៨០ ត្បូង) ពន្លឺព្រះអាទិត្យ សំណើម ពន្លឺលើផ្ទាំងសូឡា
    hillshade ០–២៥៥ ផែនទីមូលដ្ឋាន (សៀវភៅទី១)

    ee.Terrain គណនាជម្រាលដោយប្រើទំហំក្រឡាជាម៉ែត្រ ដែលវាទាញពី projection នៃរូបភាព។ រូបភាព mosaic (ឧ. GLO30 ALOS) បាត់ projection ដើម ហើយត្រូវ setDefaultProjection() ជាមុនសិន (មេរៀនទី៦)។ បើមិនដូច្នោះទេ ជម្រាលនឹងខុសទាំងស្រុង។

    ពិសោធន៍៖ ទីសណ្ឋាន និងអាងរងទឹក

    ប្ដូរផលិតផល (កម្ពស់ ជម្រាល ទិស ស្រមោល) និងទិសពន្លឺ។ ចុចលើប្រឡាយណាមួយ ដើម្បីកំណត់អាងរងទឹកនៃចំណុចនោះ ហើយមើលផ្ទៃរបស់វា។ ពិសោធន៍នេះប្រើ DEM ក្លែងធ្វើ ហើយគណនា D8 និងស្រុតទឹកក្នុងកម្មវិធីរុករករបស់អ្នក។

    ៩.៣.៣. ទិសលំហូរ និងស្រុតទឹក¶

    អានរូបនេះ

    • D8៖ ក្រឡានីមួយៗបង្ហូរទៅក្រឡាជិតខាងមួយក្នុងចំណោម ៨ ដែលជម្រាលចុះខ្លាំងបំផុត។
    • ស្រុតទឹក (flow accumulation) = ចំនួនក្រឡាដែលហូរកាត់ក្រឡានីមួយៗ។
    • ទីជ្រៅក្នុង DEM (sink) បញ្ឈប់លំហូរ៖ ត្រូវបំពេញ (fill) មុន ឬប្រើ HydroSHEDS/MERIT ដែលកែរួច។

    ទឹកហូរទៅទីទាប។ ក្បួន D8 កំណត់ថាក្រឡានីមួយៗ បង្ហូរទឹកទៅក្រឡាជិតខាងមួយក្នុងចំណោម ៨ ដែលចុះទាបខ្លាំងបំផុត (គិតជម្រាល មិនមែនកម្ពស់ទេ ព្រោះក្រឡាអង្កត់ទ្រូងឆ្ងាយជាង)។ ពីទិសលំហូរ យើងគណនា ស្រុតទឹក (flow accumulation)៖ ចំនួនក្រឡា (ឬផ្ទៃ) ខាងលើដែលហូរកាត់ក្រឡានីមួយៗ។ ក្រឡាដែលមានស្រុតទឹកខ្ពស់ គឺជាប្រឡាយ ស្ទឹង ឬទន្លេ។

    មុនគណនា D8 ត្រូវ បំពេញទីជ្រៅ (fill sinks)៖ ក្រឡាដែលទាបជាងអ្នកជិតខាងទាំងអស់ (ជាញឹកញាប់ជាកំហុស DEM) បញ្ឈប់លំហូរ។ Earth Engine មិនមានមុខងារ D8 ពេញលេញក្នុង API ទេ ប៉ុន្តែផ្ដល់ផលិតផលដែលគណនារួចពី DEM ដែលបានកែលម្អ៖

    var upa = merit.select('upa');                        // ផ្ទៃស្រុតទឹក (គម²)
var streams = upa.gt(100).selfMask();                 // ប្រឡាយ = ស្រុតទឹក > ១០០ គម²
Map.addLayer(upa.log10(), {min: 0, max: 5, palette: ['white', '#6baed6', '#08306b']}, 'log10 upa');
Map.addLayer(streams, {palette: ['#1565c0']}, 'ប្រឡាយ (> ១០០ គម²)');
var dir = merit.select('dir');                        // ទិស D8 (១ ២ ៤ ... ១២៨)

    សម្រាប់ការគណនា D8 ពីចំណុចចេញដែលអ្នកកំណត់ផ្ទាល់ (ឧ. ទំនប់មួយ) សូមនាំចេញ DEM ទៅ QGIS ហើយប្រើ GRASS r.watershed ឬ WhiteboxTools ដូចក្នុងសៀវភៅទី២។

    ៩.៣.៤. បណ្ដាញប្រឡាយ និងកម្រិតស្រុតទឹក¶

    អានរូបនេះ

    • ស្រុតទឹកមានលំដាប់ទំហំខុសគ្នាខ្លាំង (១ ដល់រាប់ម៉ឺន) ដូច្នេះបង្ហាញជា log។
    • កម្រិតស្រុតទឹក (ឧ. > ៤០០ ក្រឡា) កំណត់ថាក្រឡាណាជាប្រឡាយ៖ កម្រិតតូច = ប្រឡាយច្រើន។
    • ក្នុង Earth Engine ប្រើ WWF/HydroSHEDS/15ACC ឬ MERIT/Hydro upa ដែលគណនារួច។

    កម្រិតស្រុតទឹកដែលប្រើកំណត់ «ប្រឡាយ» ជាការសម្រេចចិត្ត៖ កម្រិតតូច (ឧ. ១ គម²) ផ្ដល់ប្រឡាយតូចៗច្រើនដែលប្រហែលជាមានទឹកតែពេលភ្លៀង។ កម្រិតធំ (ឧ. ១ ០០០ គម²) ផ្ដល់តែទន្លេធំៗ។ ប្រៀបធៀបលទ្ធផលជាមួយរូបភាពផ្កាយរណប ឬ HydroRIVERS៖

    var rivers = ee.FeatureCollection('WWF/HydroSHEDS/v1/FreeFlowingRivers')
  .filterBounds(kh);                                  // kh = ព្រំដែនកម្ពុជា
Map.addLayer(rivers.filter(ee.Filter.gte('RIV_ORD', 1)).filter(ee.Filter.lte('RIV_ORD', 5)),
  {color: '1565c0'}, 'HydroRIVERS (លំដាប់ ១–៥)');

    RIV_ORD គឺលំដាប់ទន្លេ (១ = ធំបំផុត)។ ទន្លេមេគង្គមានលំដាប់ទាបបំផុតនៅកម្ពុជា ហើយស្ទឹងតូចៗមានលំដាប់ខ្ពស់។

    ៩.៣.៥. អាងរងទឹក¶

    អានរូបនេះ

    • អាងរងទឹក = គ្រប់ក្រឡាដែលលំហូររបស់វាឆ្លងកាត់ចំណុចចេញ។
    • ក្នុង QGIS/GRASS/WhiteboxTools យើងកំណត់ចំណុចចេញ ហើយគណនាអាង។ ក្នុង Earth Engine ភាគច្រើនប្រើ HydroBASINS ដែលគណនារួច។
    • អាងរងជាអង្គភាពល្អសម្រាប់ស្ថិតិ (មេរៀនទី៤)៖ ទឹកភ្លៀងមធ្យម ផ្ទៃព្រៃ ផ្ទៃទឹកជំនន់។

    អាងរងទឹក (watershed · catchment) នៃចំណុចមួយ គឺគ្រប់ក្រឡាដែលទឹកហូរឆ្លងកាត់ចំណុចនោះ។ វាជាអង្គភាពធម្មជាតិសម្រាប់ជលសាស្ត្រ៖ ទឹកភ្លៀងលើអាងមួយ ហូរចេញតាមចំណុចចេញតែមួយ។ ក្នុង Earth Engine ប្រភពអាងរងដែលងាយប្រើបំផុតគឺ HydroBASINS៖

    var lake = ee.Geometry.Point([104.0, 12.8]);            // ក្នុងទន្លេសាប
var b6 = ee.FeatureCollection('WWF/HydroSHEDS/v1/Basins/hybas_6').filterBounds(lake);
var subs = ee.FeatureCollection('WWF/HydroSHEDS/v1/Basins/hybas_8')
  .filterBounds(b6.geometry())
  .filter(ee.Filter.bounds(b6.geometry().buffer(-2000)));   // ដកអាងដែលប៉ះតែគែម
print('អាងរងកម្រិត ៨', subs.size());
Map.addLayer(subs, {color: 'orange'}, 'អាងរងកម្រិត ៨');

    ៩.៣.៦. កម្រិតអាងរង HydroBASINS¶

    អានរូបនេះ

    • កម្រិត ៤–៥ សម្រាប់អាងទន្លេធំ (មេគង្គ)។ កម្រិត ៦–៨ សម្រាប់ទន្លេសាប និងស្ទឹងក្នុងខេត្ត។
    • អាងនីមួយៗមាន HYBAS_ID និង NEXT_DOWN សម្រាប់តាមដានលំហូរពីអាងមួយទៅអាងមួយ។
    • ព្រំអាងពី DEM ៩០ ម អាចខុសនៅវាលទំនាបរាបស្មើ ជាពិសេសជុំវិញទន្លេសាប។

    HydroBASINS មាន ១២ កម្រិត តាមប្រព័ន្ធ Pfafstetter៖ កម្រិត ១ ជាទ្វីប ហើយកម្រិតនីមួយៗបំបែកអាងមុនជាអាងតូចៗ។ Feature នីមួយៗមាន HYBAS_ID NEXT_DOWN (អាងដែលវាហូរចូល) SUB_AREA (គម²) និង UP_AREA (ផ្ទៃខាងលើសរុប)។ ជ្រើសកម្រិតតាមមាត្រដ្ឋានសំណួរ៖ កម្រិត ៥–៦ សម្រាប់អាងទន្លេធំ កម្រិត ៧–៩ សម្រាប់ស្ទឹង និងស្រុក។

    ចំណាំ៖ ព្រំអាងពី DEM ៩០ ម អាចខុសច្រើនគីឡូម៉ែត្រនៅវាលទំនាបរាបស្មើជុំវិញទន្លេសាប ដែលទឹកហូរតាមប្រឡាយធារាសាស្ត្រ និងផ្លូវ ច្រើនជាងតាមទីសណ្ឋានធម្មជាតិ។

    ៩.៣.៧. HAND៖ ទីទាបធៀបនឹងទន្លេ¶

    អានរូបនេះ

    • ទីតាំងខ្ពស់ ៤០ ម ពីនីវ៉ូសមុទ្រ អាចលិចទឹក ប្រសិនបើវាខ្ពស់ត្រឹម ២ ម ពីទន្លេជិតបំផុត។
    • HAND ធ្វើឱ្យកម្ពស់ប្រៀបធៀបបានពេញអាងទន្លេ៖ ល្អសម្រាប់ផែនទីហានិភ័យ និង mask ទឹកជំនន់ (មេរៀនទី៧)។
    • MERIT Hydro មាន HAND រួចជាស្រេច (band hnd) នៅ ៩០ ម។

    កម្ពស់ពីនីវ៉ូសមុទ្រ មិនប្រាប់ពីហានិភ័យទឹកជំនន់បានល្អទេ៖ ភូមិមួយខ្ពស់ ៤០ ម ពីនីវ៉ូសមុទ្រ អាចលិចទឹក ប្រសិនបើវាខ្ពស់ត្រឹម ២ ម ពីស្ទឹងជិតបំផុត។ HAND (Height Above Nearest Drainage) វាស់កម្ពស់នៃក្រឡានីមួយៗ ធៀបនឹងក្រឡាប្រឡាយដែលវាហូរចូល៖

    var hand = merit.select('hnd');
var lowland = hand.lt(5);                                   // ងាយលិចទឹក
Map.addLayer(hand, {min: 0, max: 30, palette: ['#08306b', '#6baed6', '#f7fbff']}, 'HAND (ម)');
var pctLow = lowland.multiply(100).rename('pct_low').reduceRegions({
  collection: subs, reducer: ee.Reducer.mean(), scale: 90});

    HAND ត្រូវបានប្រើញឹកញាប់ជាលក្ខណៈសម្រាប់ផែនទីហានិភ័យទឹកជំនន់ mask សម្រាប់ផែនទីទឹកជំនន់ SAR (មេរៀនទី៧) និងលក្ខខណ្ឌសម្រាប់ការវិភាគពហុលក្ខខណ្ឌ (មេរៀនទី១៣)។

    ៩.៤. ឧទាហរណ៍ដែលបានដោះស្រាយ៖ តារាងអាងរងទឹកទន្លេសាប¶

    ជំហានទី១៖ ជ្រើសអាងរង¶

    ប្រើកូដក្នុងផ្នែក ៩.៣.៥ ដើម្បីបានអាងរងកម្រិត ៨ ដែលស្ថិតក្នុងអាងទន្លេសាប (កម្រិត ៦)។ ពិនិត្យលើផែនទីថាអាងរងទាំងនោះគ្របជុំវិញបឹង។

    ជំហានទី២៖ រៀបចំស្រទាប់លក្ខណៈ¶

    var srtmT = ee.Terrain.products(srtm);
var rain = ee.ImageCollection('UCSB-CHG/CHIRPS/DAILY').filterDate('2014-01-01', '2024-01-01')
  .sum().divide(10).rename('rain_mm');                     // ទឹកភ្លៀងមធ្យមប្រចាំឆ្នាំ ២០១៤–២០២៣
var stack = srtmT.select(['elevation', 'slope'])
  .addBands(hand.lt(5).multiply(100).rename('pct_low'))
  .addBands(rain);

    ជំហានទី៣៖ ស្ថិតិតាមអាងរង¶

    var table = stack.reduceRegions({collection: subs, reducer: ee.Reducer.mean(), scale: 90, tileScale: 4})
  .map(function(f) { return f.set('area_km2', f.geometry().area(100).divide(1e6)); });
print(table.select(['HYBAS_ID', 'area_km2', 'elevation', 'slope', 'pct_low', 'rain_mm']).limit(10));

    នៅ scale ៩០ ម ដែលស្របនឹង MERIT Hydro ការគណនាលើអាងរងច្រើនដំណើរការលឿន។ area(100) គណនាផ្ទៃពិតជាម៉ែត្រការ៉េ (កំហុសអនុញ្ញាត ១០០ ម)។

    ជំហានទី៤៖ បកស្រាយ¶

    ក្នុង QGIS ធ្វើផែនទីពីរ៖ slope (ល្បឿនទឹកហូរ) និង pct_low (ដីងាយលិច)។ អាងរងនៅជួរភ្នំក្រវាញ (ខាងត្បូងបឹង) ទំនងជាមានជម្រាល និងទឹកភ្លៀងខ្ពស់ ខណៈអាងរងជិតបឹងមាន pct_low ខ្ពស់។ អាងរងដែលមានទាំងពីរ (ភ្លៀងច្រើនខាងលើ និងដីទំនាបខាងក្រោម) គួរតែជាអាទិភាពសម្រាប់ប្រព័ន្ធព្រមានទឹកជំនន់។

    «អាងរងកម្រិត ៨ នៃទន្លេសាប (HydroBASINS) ត្រូវបានពិពណ៌នាដោយកម្ពស់ និងជម្រាលមធ្យម (SRTM) ភាគរយដី HAND < ៥ ម (MERIT Hydro) និងទឹកភ្លៀងមធ្យមប្រចាំឆ្នាំ ២០១៤–២០២៣ (CHIRPS)។»

    អ្វីដែលយើងមិនអាចសន្និដ្ឋាន

    ព្រំអាងរង និង HAND ពី DEM ៩០ ម មិនគិតពីប្រឡាយធារាសាស្ត្រ ទំនប់ និងផ្លូវ ដែលប្ដូរលំហូរទឹកនៅវាលទំនាប។ SRTM ជា DSM ដូច្នេះកម្ពស់ក្នុងតំបន់ព្រៃខ្ពស់ជាងដីពិត។ តារាងនេះប្រៀបធៀបអាងរង មិនមែនម៉ូដែលជលសាស្ត្រដែលព្យាករណ៍លំហូរទឹកទេ។

    ៩.៥. សកម្មភាពអនុវត្ត¶

    • លំហាត់ទី៩៖ គណនាផលិតផលទីសណ្ឋាន ប្រៀបធៀប SRTM និង Copernicus GLO-30 ធ្វើផែនទីប្រឡាយពី MERIT Hydro ហើយបង្កើតតារាងលក្ខណៈអាងរងនៃខេត្តរបស់អ្នក។ មើល លំហាត់ទី៩។
    • ក្នុងថ្នាក់៖ ប្រៀបធៀបប្រឡាយពីកម្រិតស្រុតទឹក ១០ ១០០ និង ១ ០០០ គម² ជាមួយរូបភាព Sentinel-2 ហើយជ្រើសកម្រិតដែលសមបំផុតសម្រាប់ខេត្តរបស់អ្នក។

    ៩.៦. ការយល់ច្រឡំដែលត្រូវប្រុងប្រយ័ត្ន¶

    ១. «DEM បង្ហាញកម្ពស់ដី» SRTM Copernicus និង ALOS ជា DSM៖ វារួមទាំងព្រៃ និងអគារ។ DTM ពិតមានតិចណាស់នៅកម្ពុជា។

    ២. «DEM ៣០ ម តែងតែល្អជាង ៩០ ម សម្រាប់ជលសាស្ត្រ» DEM ៩០ ម ដែលបានកែលម្អ (MERIT HydroSHEDS) ជាញឹកញាប់ផ្ដល់បណ្ដាញប្រឡាយត្រឹមត្រូវជាង DSM ៣០ ម ដែលមានសំឡេងរំខាន។

    ៣. «ជម្រាលពី mosaic គណនាត្រឹមត្រូវដោយស្វ័យប្រវត្តិ» mosaic បាត់ projection ហើយ ee.Terrain នឹងគណនាខុស។ ត្រូវ setDefaultProjection។

    ៤. «ទីខ្ពស់ពីនីវ៉ូសមុទ្រ មិនលិចទឹក» ហានិភ័យអាស្រ័យលើកម្ពស់ធៀបនឹងទន្លេជិតបំផុត (HAND) មិនមែនកម្ពស់ដាច់ខាតទេ។

    ៥. «ព្រំអាងរងពី DEM គឺជាព្រំពិត» នៅវាលទំនាប ប្រឡាយ ទំនប់ និងផ្លូវ ប្ដូរលំហូរ។ ព្រំអាងរងត្រូវផ្ទៀងផ្ទាត់ជាមួយផែនទីធារាសាស្ត្រ និងចំណេះដឹងក្នុងស្រុក។

    ៩.៧. សេចក្ដីសង្ខេបមេរៀន¶

    DEM ក្នុង Earth Engine (SRTM Copernicus ALOS MERIT) ភាគច្រើនជា DSM ៣០–៩០ ម៉ែត្រ។ ee.Terrain ផ្ដល់ជម្រាល ទិស និងស្រមោលភ្នំ លុះត្រាតែរូបភាពមាន projection ត្រឹមត្រូវ។ ទឹកហូរតាមទិស D8 ទៅក្រឡាជិតខាងដែលទាបជាងគេ ហើយស្រុតទឹកកំណត់បណ្ដាញប្រឡាយ ដោយកម្រិតដែលយើងជ្រើស។

    HydroBASINS ផ្ដល់អាងរងជា ១២ កម្រិត ដែលជាអង្គភាពល្អសម្រាប់ស្ថិតិ ហើយ MERIT Hydro ផ្ដល់ស្រុតទឹក និង HAND រួចជាស្រេច។ HAND វាស់ភាពទាបធៀបនឹងប្រឡាយជិតបំផុត ហើយជាលក្ខណៈសំខាន់សម្រាប់ហានិភ័យទឹកជំនន់។ នៅវាលទំនាបកម្ពុជា កំហុស DEM និងហេដ្ឋារចនាសម្ព័ន្ធទឹក ធ្វើឱ្យលទ្ធផលត្រូវផ្ទៀងផ្ទាត់ជានិច្ច។

    គំនិតស្នូលនៃមេរៀន

    ទឹកហូរតាមទីសណ្ឋាន ហើយអាងរងទឹកជាអង្គភាពធម្មជាតិនៃវដ្ដទឹក។ ប៉ុន្តែនៅប្រទេសដែលរាបស្មើដូចកម្ពុជា ភាពត្រឹមត្រូវនៃ DEM និងការកែប្រែដោយមនុស្ស សំខាន់ដូចក្បួនដោះស្រាយដែរ។

    ៩.៨. ពាក្យគន្លឹះខ្មែរ–អង់គ្លេស¶

    ពាក្យខ្មែរ ពាក្យអង់គ្លេស អត្ថន័យខ្លី
    គំរូកម្ពស់ឌីជីថល DEM រ៉ាស្ទ័រកម្ពស់
    គំរូផ្ទៃ/គំរូដី DSM / DTM ផ្ទៃខាងលើ (ព្រៃ អគារ) / ដីទទេ
    ជម្រាល · ទិស Slope · Aspect មុំ និងទិសនៃផ្ទៃ
    ស្រមោលភ្នំ Hillshade ស្រមោលក្លែងធ្វើពីពន្លឺ
    ទិសលំហូរ Flow direction (D8) ក្រឡាដែលទឹកហូរទៅ
    ស្រុតទឹក Flow accumulation ផ្ទៃខាងលើដែលហូរកាត់
    អាងរងទឹក Watershed / Catchment តំបន់ដែលហូរចេញតាមចំណុចមួយ
    កម្ពស់ពីលើប្រឡាយជិតបំផុត HAND កម្ពស់ធៀបនឹងប្រឡាយជិតបំផុត

    ៩.៩. សំណួររំលឹក និងវាយតម្លៃ¶

    ក. សំណួរយល់ដឹង¶

    1. ភាពខុសគ្នារវាង DSM និង DTM? ហេតុអ្វីវាសំខាន់នៅតំបន់ព្រៃ?
    2. ហេតុអ្វី DEM ៩០ ម ដែលបានកែលម្អ អាចល្អជាង DSM ៣០ ម សម្រាប់ប្រឡាយ?
    3. ពន្យល់ D8 និងស្រុតទឹក ដោយប្រើក្រឡា ៣ × ៣ ជាឧទាហរណ៍។
    4. ហេតុអ្វីត្រូវបំពេញទីជ្រៅមុនគណនាទិសលំហូរ?
    5. ហេតុអ្វី HAND ល្អជាងកម្ពស់ដាច់ខាត សម្រាប់ហានិភ័យទឹកជំនន់?

    ខ. សំណួរអនុវត្តគោលគំនិត¶

    1. សរសេរកូដគណនាភាគរយនៃខេត្តរបស់អ្នក ដែលមានជម្រាល > ១៥°។
    2. ប្រឡាយពីកម្រិត upa > ១ គម² មានច្រើនពេក។ តើអ្នកនឹងប្ដូរអ្វី ហើយផ្ទៀងផ្ទាត់ដោយរបៀបណា?
    3. សរសេរកូដរកអាងរងកម្រិត ៧ ដែលផ្ទុកទីរួមខេត្តរបស់អ្នក ហើយគណនាផ្ទៃរបស់វា។
    4. អាងរងមួយមាន pct_low = ៦០% និងជម្រាលមធ្យម ០,៨°។ តើវាមានលក្ខណៈអ្វី ហើយហានិភ័យអ្វី?
    5. ស្នើរបៀបរួមបញ្ចូលមេរៀនទី៧ ៨ និង ៩ ដើម្បីផែនទីហានិភ័យទឹកជំនន់តាមអាងរង។

    គ. កិច្ចការខ្លី¶

    ប្រៀបធៀបជម្រាលពី SRTM និង Copernicus GLO-30 ក្នុងតំបន់ព្រៃមួយ និងវាលស្រែមួយនៃខេត្តរបស់អ្នក។ សរសេរកថាខណ្ឌមួយពន្យល់ភាពខុសគ្នា ដោយយោងលើ DSM ធៀបនឹង DTM។

    កំណត់សម្គាល់សម្រាប់គ្រូបង្រៀន¶

    ខ. ចំណាំអំពីទិន្នន័យ¶

    • រូប D8 ស្រុតទឹក អាងរង និងពិសោធន៍ ប្រើ DEM ក្លែងធ្វើ ដើម្បីឱ្យការគណនាធ្វើបានក្នុងកម្មវិធីរុករក។ លំហាត់ទី៩ ប្រើ DEM ពិតក្នុង Earth Engine។
    • ប្រសិនបើនិស្សិតចង់កំណត់អាងរងពីចំណុចចេញផ្ទាល់ខ្លួន សូមណែនាំ GRASS r.watershed ក្នុង QGIS ជាមួយ DEM ដែលនាំចេញពីលំហាត់ទី៩។

    គ. ការភ្ជាប់ទៅមេរៀនបន្ទាប់¶

    ជំពូកទី៣ បានបង្ហាញពីទឹក៖ ទឹកច្រើនពេក ទឹកតិចពេក និងផ្លូវដែលទឹកហូរ។ ជំពូកបន្ទាប់ផ្ដោតលើដី និងមនុស្ស៖ ព្រៃឈើ កសិកម្ម និងទីក្រុង ដែលប្ដូរលឿនបំផុតនៅកម្ពុជាក្នុងរយៈពេល ២០ ឆ្នាំចុងក្រោយ។

    តើកម្ពុជាបាត់បង់ព្រៃឈើប៉ុន្មាន នៅទីណា និងពេលណា ហើយតំបន់ការពារបានជួយការពារវាឬទេ?
