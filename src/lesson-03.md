# មេរៀនទី៣៖ Image Collection ការត្រង ការលុបពពក និង Composite

!!! info "ព័ត៌មានមេរៀន"
    **រយៈពេលបង្រៀនដែលស្នើ៖ ៣ ម៉ោង** · មេរៀនទី ៣ ក្នុងចំណោម ១៥

នៅកម្ពុជា រូបភាពផ្កាយរណបអុបទិកស្អាតទាំងស្រុងមួយផ្ទាំង គឺជារឿងកម្រ ជាពិសេសចាប់ពីខែឧសភាដល់ខែតុលា។ ក្នុងសៀវភៅទី៣ យើងដោះស្រាយបញ្ហានេះដោយស្វែងរកថ្ងៃមួយដែលពពកតិច។ ជាមួយ Earth Engine យើងមានជម្រើសប្រសើរជាង៖ **ផ្គុំផ្នែកស្អាតពីរូបភាពរាប់សិបផ្ទាំង** ទៅជារូបភាពថ្មីមួយ ដែលគ្មានពពក។ រូបភាពនេះហៅថា **composite**។

មេរៀននេះបង្រៀនជំហានទាំងបី ដែលស្ទើរតែគ្រប់ស្គ្រីប Earth Engine ចាប់ផ្ដើមជាមួយ៖ ត្រង ImageCollection លុបពពកតាមក្រឡា និងបង្កើត composite ដោយ reducer ដែលសមស្រប។

## ៣.១. គោលបំណងសិក្សា

បន្ទាប់ពីបញ្ចប់មេរៀន និស្សិតអាច៖

1. ពិពណ៌នា ImageCollection ជាគូបពេលវេលា (ទីតាំង × ពេលវេលា)។
2. ត្រង collection តាមទីតាំង កាលបរិច្ឆេទ និង metadata ហើយពិនិត្យចំនួនដែលនៅសល់។
3. លុបពពក និងស្រមោលពពក ដោយប្រើ SCL (Sentinel-2) និង QA_PIXEL (Landsat)។
4. បង្កើត composite ដោយ median() និង qualityMosaic() ហើយពន្យល់ភាពខុសគ្នា។
5. ជ្រើសរយៈពេល និងវិធី composite ដែលសមស្របនឹងសំណួរ និងរដូវនៅកម្ពុជា។

មន្ទីររៀបចំដែនដីខេត្តមួយ ត្រូវការ **រូបភាពមូលដ្ឋានពណ៌ពិតថ្មីបំផុត** សម្រាប់ខេត្តទាំងមូល ដើម្បីធ្វើការបកស្រាយដោយភ្នែក។ ខេត្តនេះគ្របលើ tile Sentinel-2 បួន ហើយក្នុងរយៈពេល ៦ ខែចុងក្រោយ គ្មានថ្ងៃណាដែល tile ទាំងបួនស្អាតព្រមគ្នាទេ។ រូបភាពដែលស្អាតបំផុតនៃ tile នីមួយៗ មកពីខែផ្សេងៗគ្នា ហើយពណ៌ខុសគ្នា។

## ៣.៣. ទ្រឹស្ដីស្នូល

### ៣.៣.១. ImageCollection ជាគូបពេលវេលា

{{FIG:a03-cube}}

ImageCollection អាចគិតជាគូបបីវិមាត្រ៖ **x និង y** (ទីតាំង) និង **t** (ពេលវេលា)។ នៅក្រឡានីមួយៗ យើងមានស៊េរីតម្លៃមួយ៖ តម្លៃមួយក្នុងរូបភាពនីមួយៗដែលគ្របក្រឡានោះ។ ការយល់ឃើញនេះសំខាន់ ព្រោះ reducer តាមពេលវេលា (median min max) ធ្វើការ **ក្រឡាទល់ក្រឡា** និង **ក្រុមរលកទល់ក្រុមរលក** លើស៊េរីនោះ។

### ៣.៣.២. ការត្រងជាជំហាន

{{FIG:a03-funnel}}

ការត្រងមានបីប្រភេទ៖

```javascript
var aoi = ee.FeatureCollection('FAO/GAUL/2015/level1')
  .filter(ee.Filter.eq('ADM1_NAME', 'Kampong Chhnang'));

var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
  .filterBounds(aoi)                                     // ១. តាមទីតាំង
  .filterDate('2024-01-01', '2024-05-01')                // ២. តាមកាលបរិច្ឆេទ
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 40));  // ៣. តាម metadata
print('ចំនួនរូបភាព', s2.size());
```

`filterBounds` និង `filterDate` ធ្វើមុន ព្រោះវាកាត់បន្ថយចំនួនពីរាប់លានទៅរាប់រយយ៉ាងលឿន។ តម្រងពពកតាម metadata គួរតែ **ធូរ** (ឧ. < ៤០%) ព្រោះការលុបពពកតាមក្រឡានឹងធ្វើការងារលម្អិតបន្ទាប់។ តម្រងតឹងពេក (< ៥%) អាចទុករូបភាពតិចពេកសម្រាប់ composite។

### ៣.៣.៣. ការលុបពពកតាមក្រឡា

{{FIG:a03-qa}}

ផលិតផល Level-2 មានក្រុមរលកគុណភាព ដែលប្រាប់ថាក្រឡានីមួយៗជាពពក ស្រមោល ឬផ្ទៃស្អាត។ **Sentinel-2 L2A** មានក្រុមរលក `SCL` (Scene Classification Layer)៖

```javascript
function maskS2(img) {
  var scl = img.select('SCL');
  // ៣ ស្រមោលពពក · ៨ ៩ ពពក · ១០ cirrus
  var clear = scl.neq(3).and(scl.neq(8)).and(scl.neq(9)).and(scl.neq(10));
  return img.updateMask(clear).divide(10000)          // ការចាំងផ្លាត ០–១
            .copyProperties(img, ['system:time_start']);
}
var s2clean = s2.map(maskS2);
```

**Landsat Collection 2** ប្រើប៊ីតក្នុង `QA_PIXEL`៖ ប៊ីត ៣ = ពពក ប៊ីត ៤ = ស្រមោលពពក។ ត្រូវប្រើ `bitwiseAnd()` ដើម្បីអានប៊ីតនីមួយៗ៖

```javascript
function maskL8(img) {
  var qa = img.select('QA_PIXEL');
  var clear = qa.bitwiseAnd(1 << 3).eq(0)       // មិនមែនពពក
              .and(qa.bitwiseAnd(1 << 4).eq(0)); // មិនមែនស្រមោល
  var sr = img.select('SR_B.').multiply(0.0000275).add(-0.2);   // មាត្រដ្ឋាន C2 L2
  return img.addBands(sr, null, true).updateMask(clear);
}
```

`updateMask()` មិនលុបក្រឡាទេ៖ វាសម្គាល់វាថា «គ្មានទិន្នន័យ» ដូច្នេះ reducer ក្រោយមកមិនរាប់ក្រឡានោះ។ របាំងពពកទាំងនេះមិនល្អឥតខ្ចោះ៖ ពពកស្ដើង និងស្រមោលជាប់គែមពពក តែងរត់ចួល។ Sentinel-2 ក៏មាន **Cloud Score+** (`GOOGLE/CLOUD_SCORE_PLUS/V1/S2_HARMONIZED`) ដែលត្រឹមត្រូវជាង SCL សម្រាប់តំបន់ត្រូពិច ហើយនឹងប្រើក្នុងលំហាត់។

### ៣.៣.៤. Composite ដោយ reducer តាមពេលវេលា

{{FIG:a03-dates}}

{{FIG:a03-reducers}}

នៅពេលមាន collection ដែលបានលុបពពករួច reducer មួយបម្លែងវាទៅជារូបភាពមួយ៖

| Reducer | ជ្រើសអ្វីនៅក្រឡានីមួយៗ | ប្រើពេលណា |
|---|---|---|
| `median()` | តម្លៃកណ្ដាល (តាមក្រុមរលកនីមួយៗ) | រូបភាពមូលដ្ឋាន ស្អាត និងស្ថិរ |
| `mean()` | មធ្យម | ទិន្នន័យស្អាតរួចហើយ (ឧ. សីតុណ្ហភាព) |
| `min()` / `max()` | តម្លៃទាប/ខ្ពស់បំផុត | សន្ទស្សន៍ (NDVI max · NDWI max) មិនមែនពណ៌ |
| `qualityMosaic('NDVI')` | រូបភាពដែល NDVI ខ្ពស់បំផុត | រុក្ខជាតិបៃតងបំផុត · ពណ៌ស៊ីគ្នា |
| `mosaic()` | រូបភាពចុងក្រោយដែលមានទិន្នន័យ | បំពេញចន្លោះតាមអាទិភាព |

`median()` ជាជម្រើសលំនាំដើមល្អបំផុត ព្រោះវាធន់នឹងតម្លៃខុសប្រក្រតី៖ ពពកដែលរត់ចួល (ភ្លឺខ្លាំង) និងស្រមោល (ងងឹតខ្លាំង) ស្ថិតនៅចុងទាំងពីរនៃស៊េរី ហើយមិនត្រូវបានជ្រើសទេ លុះត្រាតែវាលើសពាក់កណ្ដាលនៃរូបភាព។

### ៣.៣.៥. median ធៀបនឹង qualityMosaic

{{FIG:a03-quality}}

`median()` ធ្វើការ **ក្រុមរលកម្ដងមួយ**៖ តម្លៃក្រហមអាចមកពីថ្ងៃមួយ ហើយតម្លៃ NIR មកពីថ្ងៃផ្សេង។ សម្រាប់ការមើល និងការចាត់ថ្នាក់ទូទៅ នេះមិនជាបញ្ហាទេ ប៉ុន្តែវាអាចបង្កើត «សញ្ញាណ» ដែលមិនធ្លាប់មាននៅថ្ងៃណាមួយពិតប្រាកដ។ `qualityMosaic()` ជ្រើស **រូបភាពទាំងមូលមួយ** ក្នុងក្រឡានីមួយៗ តាមលក្ខណៈវិនិច្ឆ័យ (ឧ. NDVI ខ្ពស់បំផុត) ដូច្នេះក្រុមរលកទាំងអស់មកពីថ្ងៃដូចគ្នា៖

```javascript
var withNdvi = s2clean.map(function(img) {
  return img.addBands(img.normalizedDifference(['B8', 'B4']).rename('NDVI'));
});
var greenest = withNdvi.qualityMosaic('NDVI');
var medianC  = s2clean.median();
Map.addLayer(medianC.clip(aoi), {bands: ['B4','B3','B2'], min: 0, max: 0.3}, 'median');
Map.addLayer(greenest.clip(aoi), {bands: ['B8','B4','B3'], min: 0, max: 0.4}, 'greenest pixel');
```

### ៣.៣.៦. ជ្រើសរយៈពេល composite នៅកម្ពុជា

រយៈពេល composite គឺជាការសម្របសម្រួល៖ **រយៈពេលវែង** ផ្ដល់ក្រឡាស្អាតច្រើន ប៉ុន្តែលាយស្ថានភាពខុសៗគ្នា (ស្រែដាំ និងស្រែច្រូតរួច ទឹកឡើង និងទឹកស្រក)។ **រយៈពេលខ្លី** រក្សាស្ថានភាពតែមួយ ប៉ុន្តែអាចមានចន្លោះ (ក្រឡាគ្មានទិន្នន័យ)។

| គោលបំណង | រយៈពេលណែនាំ |
|---|---|
| រូបភាពមូលដ្ឋានពណ៌ពិត | រដូវប្រាំង (វិច្ឆិកា–មេសា) ឆ្នាំតែមួយ |
| គម្របដីប្រចាំឆ្នាំ | composite ពីរ៖ រដូវប្រាំង + រដូវវស្សា (រួម Sentinel-1) |
| ស្រែ និងដំណាំ | ប្រចាំខែ ឬ ១៦ ថ្ងៃ (មេរៀនទី៥ និង ១១) |
| ទឹកជំនន់ | ជុំវិញថ្ងៃកើតហេតុ + SAR (មេរៀនទី៧) |

!!! warning "Composite មិនមែនជារូបភាពពីថ្ងៃតែមួយទេ"
    តែងតែសរសេររយៈពេល និង reducer ក្នុងចំណងជើងផែនទី (ឧ. «Sentinel-2 median · វិច្ឆិកា ២០២៣–មេសា ២០២៤»)។ ការផ្លាស់ប្ដូរដែលកើតក្នុងរយៈពេល composite (ឧ. ការឈូសឆាយ) អាចបង្ហាញមិនពេញលេញ។

---

## ៣.៤. ឧទាហរណ៍ដែលបានដោះស្រាយ៖ រូបភាពមូលដ្ឋានខេត្តកំពង់ឆ្នាំង

### ជំហានទី១៖ កំណត់តំបន់ និងរយៈពេល

រដូវប្រាំង ១ វិច្ឆិកា ២០២៣ ដល់ ៣០ មេសា ២០២៤ ខេត្តកំពង់ឆ្នាំង (GAUL level 1)។

### ជំហានទី២៖ ត្រង និងលុបពពក

```javascript
var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
  .filterBounds(aoi).filterDate('2023-11-01', '2024-05-01')
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 40))
  .map(maskS2);
print('រូបភាពប្រើ', s2.size());
```

### ជំហានទី៣៖ Composite ហើយពិនិត្យ

```javascript
var base = s2.median().clip(aoi);
Map.addLayer(base, {bands: ['B4','B3','B2'], min: 0, max: 0.3}, 'មូលដ្ឋាន');
// ពិនិត្យចំនួនក្រឡាស្អាតដែលប្រើក្នុងក្រឡានីមួយៗ
var nClear = s2.select('B4').count().clip(aoi);
Map.addLayer(nClear, {min: 0, max: 30, palette: ['red', 'yellow', 'green']}, 'ចំនួនការសង្កេតស្អាត');
```

### ជំហានទី៤៖ បកស្រាយ

ស្រទាប់ `count()` បង្ហាញថាក្រឡាភាគច្រើនមានការសង្កេតស្អាត ២០–៣០ ប៉ុន្តែតំបន់ខ្លះនៅជិតភ្នំមានតិចជាង ៥។ តំបន់ទាំងនោះទុកចិត្តតិចជាង ព្រោះ median ពីតម្លៃតិចងាយរងឥទ្ធិពលពពកដែលរត់ចួល។

### ជំហានទី៥៖ សេចក្ដីសន្និដ្ឋាន

> «Composite median រដូវប្រាំង ២០២៣–២០២៤ ពី Sentinel-2 L2A ដែលបានលុបពពកដោយ SCL ផ្ដល់រូបភាពមូលដ្ឋានស្អាតសម្រាប់ខេត្តទាំងមូល។ ស្រទាប់ចំនួនការសង្កេតត្រូវភ្ជាប់ជាមួយ ដើម្បីបង្ហាញតំបន់ដែលទុកចិត្តតិច។»

!!! warning "អ្វីដែលយើងមិនអាចសន្និដ្ឋាន"
    រូបភាពនេះមិនតំណាងថ្ងៃណាមួយទេ។ វាមិនអាចប្រើដើម្បីប្រាប់ថា «ស្រែនេះត្រូវបានច្រូតនៅថ្ងៃ X» ឬវាស់ការផ្លាស់ប្ដូរក្នុងរយៈពេល ៦ ខែនោះទេ។

## ៣.៥. សកម្មភាពអនុវត្ត

- **លំហាត់ទី៣**៖ បង្កើត composite រដូវប្រាំង និងរដូវវស្សានៃខេត្តរបស់អ្នក ប្រៀបធៀប median និង qualityMosaic ហើយនាំចេញទៅ Google Drive។ មើល [លំហាត់ទី៣](../workbook/lab-03.md)។
- **ក្នុងថ្នាក់**៖ ក្រុមនីមួយៗធ្វើ composite សម្រាប់រដូវខុសគ្នា ហើយប្រៀបធៀបចំនួនការសង្កេតស្អាត។

## ៣.៦. ការយល់ច្រឡំដែលត្រូវប្រុងប្រយ័ត្ន

**១. «តម្រង CLOUDY_PIXEL_PERCENTAGE លុបពពករួចហើយ»**
វាទុក ឬបោះរូបភាពទាំងមូល។ ក្រឡាពពកក្នុងរូបភាពដែលនៅសល់ ត្រូវតែ mask ដាច់ដោយឡែក។

**២. «max() ផ្ដល់រូបភាពភ្លឺស្អាតជាង»**
max() ជ្រើសពពក ព្រោះពពកភ្លឺបំផុត។ ប្រើ max() សម្រាប់សន្ទស្សន៍ ដូចជា NDVI ប៉ុណ្ណោះ។

**៣. «Composite គឺជារូបភាពពីថ្ងៃដែលស្អាតបំផុត»**
ក្រឡាជិតគ្នាអាចមកពីថ្ងៃខុសគ្នា ហើយ median អាចលាយក្រុមរលកពីថ្ងៃខុសគ្នាក្នុងក្រឡាតែមួយ។

**៤. «រយៈពេលកាន់តែវែង composite កាន់តែល្អ»**
រយៈពេលវែងលាយរដូវ និងលាក់ការផ្លាស់ប្ដូរ។

**៥. «updateMask លុបទិន្នន័យជាអចិន្ត្រៃយ៍»**
វាគ្រាន់តែសម្គាល់ក្រឡាថាគ្មានទិន្នន័យក្នុងការគណនានេះ។ ទិន្នន័យដើមក្នុងកាតាឡុកមិនប្ដូរទេ។

## ៣.៧. សេចក្ដីសង្ខេបមេរៀន

ImageCollection ជាគូបពេលវេលា ដែលក្រឡានីមួយៗមានស៊េរីតម្លៃ។ ការត្រងតាមទីតាំង កាលបរិច្ឆេទ និង metadata កាត់បន្ថយចំនួនរូបភាព ហើយការលុបពពកតាមក្រឡា (SCL · QA_PIXEL · Cloud Score+) ដកតម្លៃមិនល្អចេញ។

Reducer តាមពេលវេលាបម្លែងស៊េរីទៅជារូបភាពមួយ៖ median() ជាជម្រើសលំនាំដើមដែលធន់នឹងពពកដែលរត់ចួល ហើយ qualityMosaic() រក្សាក្រុមរលកពីថ្ងៃដូចគ្នា។ រយៈពេល composite ត្រូវជ្រើសតាមសំណួរ និងរដូវ ហើយត្រូវរាយការណ៍ជាមួយចំនួនការសង្កេតស្អាត។

!!! abstract "គំនិតស្នូលនៃមេរៀន"
    Composite ប្ដូរសំណួរពី «ថ្ងៃណាស្អាត?» ទៅជា «ក្រឡានីមួយៗ តម្លៃអ្វីដែលតំណាងរយៈពេលនេះបានល្អបំផុត?» ហើយចម្លើយអាស្រ័យលើ reducer ដែលអ្នកជ្រើស។

## ៣.៩. សំណួររំលឹក និងវាយតម្លៃ

### ក. សំណួរយល់ដឹង

1. ពិពណ៌នា ImageCollection ជាគូបពេលវេលា ហើយពន្យល់ថា reducer ធ្វើការនៅកម្រិតណា។
2. ហេតុអ្វីត្រូវ filterBounds និង filterDate មុន filter តាម metadata?
3. រាយតម្លៃ SCL ដែលត្រូវ mask ហើយពន្យល់ថាតម្លៃនីមួយៗមានន័យអ្វី។
4. ហេតុអ្វី median() ធន់នឹងពពកដែលរត់ចួល ជាង mean()?
5. ភាពខុសគ្នារវាង median() និង qualityMosaic('NDVI')?

### ខ. សំណួរអនុវត្តគោលគំនិត

6. សរសេរមុខងារ mask សម្រាប់ Landsat 9 C2 L2 ដែលលុបទាំងពពក cirrus (ប៊ីត ២) ផង។
7. ក្រឡាមួយមានតម្លៃក្រហម [០,០៥ ០,៤២ ០,០៦ ០,០៧ ០,៣៨]។ គណនា median និង mean ហើយពន្យល់ថាមួយណាតំណាងផ្ទៃដីល្អជាង។
8. ហេតុអ្វីមិនគួរប្រើ max() សម្រាប់ composite ពណ៌ពិត ប៉ុន្តែអាចប្រើសម្រាប់ NDVI?
9. ស្នើរយៈពេល composite សម្រាប់ផែនទីផ្ទៃទឹកអតិបរមាប្រចាំឆ្នាំនៃទន្លេសាប ហើយពន្យល់។
10. ស្រទាប់ count() បង្ហាញតម្លៃ ២ នៅតំបន់មួយ។ តើអ្នកនឹងធ្វើអ្វី?

### គ. កិច្ចការខ្លី

បង្កើត composite median សម្រាប់ខេត្តរបស់អ្នកក្នុងរយៈពេលបីផ្សេងគ្នា (១ ខែ ៣ ខែ ៦ ខែ)។ ថតរូបអេក្រង់ ហើយសរសេរកថាខណ្ឌមួយប្រៀបធៀបចន្លោះ ពពកដែលរត់ចួល និងការលាយរដូវ។
